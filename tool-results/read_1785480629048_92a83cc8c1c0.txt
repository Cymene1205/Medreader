   354→          try {

   361→              setFigures(figRes.figures as Figure[]);
   362→            }

   373→              Array.isArray(figRes.figures) &&
   374→              figRes.figures.length > 0 &&
   375→              figRes.figures.every((f: Figure) => f.question);

   381→            }
   382→          } catch (e) {
   383→            console.warn("[poll] figures/analyze fetch failed:", e);
   384→          }
   385→          setTimeout(poll, 4000);
   386→        };
   387→        setTimeout(poll, 3000);
   388→      }

   390→    []
