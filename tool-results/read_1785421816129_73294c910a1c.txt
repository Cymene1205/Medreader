   143→// ---------- Helpers ----------
   144→function formatDateShort(iso: string | null): string {
   145→  if (!iso) return "—";
   146→  const d = new Date(iso);
   147→  if (Number.isNaN(d.getTime())) return "—";
   148→  return `${d.getMonth() + 1}/${d.getDate()}`;
   149→}
   150→
   151→function formatDateTime(iso: string | null): string {
   152→  if (!iso) return "—";
   153→  const d = new Date(iso);
   154→  if (Number.isNaN(d.getTime())) return "—";
   155→  const y = d.getFullYear();
   156→  const m = String(d.getMonth() + 1).padStart(2, "0");
   157→  const day = String(d.getDate()).padStart(2, "0");
   158→  const hh = String(d.getHours()).padStart(2, "0");
   159→  const mm = String(d.getMinutes()).padStart(2, "0");
   160→  return `${y}-${m}-${day} ${hh}:${mm}`;
   161→}
   162→
   163→function datePart(iso: string | null): string {
   164→  if (!iso) return "";
   165→  // ISO strings start with YYYY-MM-DD regardless of T or space separator.
   166→  return iso.slice(0, 10);
   167→}
   168→
   169→/** Format a token count with thousands separators + k/M suffix. */
   170→function formatTokens(n: number): string {
   171→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(2)}M`;
   172→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}k`;
   173→  return String(n);
   174→}
   175→
   176→/** Format a CNY cost: large numbers use 万, small numbers use 4 decimals. */
   177→function formatCny(n: number): string {
   178→  if (n >= 10_000) return `${(n / 10_000).toFixed(2)}万`;
   179→  if (n >= 1) return n.toFixed(2);
   180→  return n.toFixed(4);
   181→}
