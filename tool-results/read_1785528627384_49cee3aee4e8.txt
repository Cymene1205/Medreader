   144→// ---------- Helpers ----------
   145→function formatDateShort(iso: string | null): string {
   146→  if (!iso) return "—";
   147→  const d = new Date(iso);
   148→  if (Number.isNaN(d.getTime())) return "—";
   149→  return `${d.getMonth() + 1}/${d.getDate()}`;
   150→}
   151→
   152→function formatDateTime(iso: string | null): string {
   153→  if (!iso) return "—";
   154→  const d = new Date(iso);
   155→  if (Number.isNaN(d.getTime())) return "—";
   156→  const y = d.getFullYear();
   157→  const m = String(d.getMonth() + 1).padStart(2, "0");
   158→  const day = String(d.getDate()).padStart(2, "0");
   159→  const hh = String(d.getHours()).padStart(2, "0");
   160→  const mm = String(d.getMinutes()).padStart(2, "0");
   161→  return `${y}-${m}-${day} ${hh}:${mm}`;
   162→}
   163→
   164→function datePart(iso: string | null): string {
   165→  if (!iso) return "";
   166→  // ISO strings start with YYYY-MM-DD regardless of T or space separator.
   167→  return iso.slice(0, 10);
   168→}
   169→
   170→/** Format a token count with thousands separators + k/M suffix. */
   171→function formatTokens(n: number): string {
   172→  if (n >= 1_000_000) return `${(n / 1_000_000).toFixed(2)}M`;
   173→  if (n >= 1_000) return `${(n / 1_000).toFixed(1)}k`;
   174→  return String(n);
   175→}
   176→
   177→/** Format a CNY cost: large numbers use 万, small numbers use 4 decimals. */
   178→function formatCny(n: number): string {
   179→  if (n >= 10_000) return `${(n / 10_000).toFixed(2)}万`;
   180→  if (n >= 1) return n.toFixed(2);
   181→  return n.toFixed(4);
   182→}
