     1→"use client";
     2→
     3→import { useEffect, useRef, useState, useCallback } from "react";
     4→import {
     5→  FileText,
     6→  Loader2,
     7→  ZoomIn,
     8→  ZoomOut,
     9→  ChevronLeft,
    10→  ChevronRight,
    11→  MousePointer2,
    12→  Square,
    13→} from "lucide-react";
    14→import { Button } from "@/components/ui/button";
    15→import { cn } from "@/lib/utils";
    16→
    17→type Props = {
    18→  fileData: ArrayBuffer | null;
    19→  fileName?: string;
    20→  onTextSelect?: (text: string) => void;
    21→  onImageSelect?: (imageBase64: string) => void;
    22→  /** keyword to highlight & scroll to (from outline click) */
    23→  highlightToken?: { quote: string; keywords: string[]; nonce: number } | null;
    24→  /** Direct page jump — when this changes (via nonce), scroll the PDF to
    25→   *  the given page (1-indexed). Used by "跳到原图 p.N" buttons. */
    26→  jumpToPage?: { pageIndex: number; nonce: number } | null;
    27→};
    28→
    29→type PageInfo = {
    30→  pageNum: number;
    31→  width: number;
    32→  height: number;
    33→  textItems: { str: string; transform: number[]; width: number; height: number }[];
    34→};
    35→
    36→export default function PdfViewer({
    37→  fileData,
    38→  fileName,
    39→  onTextSelect,
    40→  onImageSelect,
    41→  highlightToken,
    42→  jumpToPage,
    43→}: Props) {
    44→  const containerRef = useRef<HTMLDivElement>(null);
    45→  const pagesRef = useRef<HTMLDivElement[]>([]);
    46→  const [pdfDoc, setPdfDoc] = useState<any>(null);
    47→  const [numPages, setNumPages] = useState(0);
    48→  const [loading, setLoading] = useState(false);
    49→  const [scale, setScale] = useState(1.2);
    50→  const [mode, setMode] = useState<"text" | "image">("text");
    51→  const [pages, setPages] = useState<PageInfo[]>([]);
    52→  const [error, setError] = useState<string | null>(null);
    53→  const [libReady, setLibReady] = useState(false);
    54→  const pdfjsLibRef = useRef<any>(null);
    55→
    56→  // Load pdfjs-dist dynamically (client-only)
    57→  useEffect(() => {
    58→    let mounted = true;
    59→    let cancelled = false;
    60→    (async () => {
    61→      try {
    62→        // ── Step 1: load the pdfjs-dist library itself ──────────────────
    63→        let lib: any;
    64→        try {
    65→          lib = await import("pdfjs-dist");
    66→        } catch (importErr) {
    67→          console.error("[pdf-viewer] import('pdfjs-dist') failed:", importErr);
    68→          if (!cancelled) {
    69→            setError(
    70→              `pdfjs-dist 库加载失败：${importErr instanceof Error ? importErr.message : String(importErr)}。请检查网络或刷新页面重试。`
    71→            );
    72→          }
    73→          return;
    74→        }
    75→        if (cancelled) return;
    76→
    77→        // ── Step 2: locate a worker ──────────────────────────────────────
    78→        // pdf.worker.js is an ESM module (file ends with `export{...}`).
    79→        // Spawning it as a classic Worker via `new Worker(url)` throws
    80→        // SyntaxError. We must spawn with `{ type: "module" }`.
    81→        //
    82→        // Strategy:
    83→        //   1. Try to create a module Worker from same-origin URL
    84→        //      (/pdf.worker.js or /api/pdf-worker). Pass it to pdf.js
    85→        //      via GlobalWorkerOptions.workerPort — this bypasses
    86→        //      pdf.js's own worker-spawning logic and works reliably
    87→        //      across dev/prod/proxy setups.
    88→        //   2. If that fails, fall back to pdf.js's workerSrc + Blob URL
    89→        //      (let pdf.js spawn the worker itself).
    90→        //   3. Last resort: CDN workerSrc (cdnjs). Will fail behind GFW
    91→        //      but at least initializes.
    92→        //   4. If everything fails, set workerSrc to "" — pdf.js falls
    93→        //      back to fake-worker mode (main thread, slow but works).
    94→        const sameOriginCandidates = ["/pdf.worker.js", "/api/pdf-worker"];
    95→
    96→        let workerReady = false;
    97→
    98→        // ── Approach A: try `new Worker(url, { type: "module" })` ──────
    99→        // This is the most reliable way to load an ESM worker. If the URL
   100→        // returns 404 or a non-JS body, the Worker constructor may still
   101→        // succeed but fail at message-handshake time — so we also do a
   102→        // HEAD-like GET first to validate the URL exists and is JS.
   103→        for (const url of sameOriginCandidates) {
   104→          if (workerReady) break;
   105→          try {
   106→            // Quick HEAD-like check (we just look at the headers).
   107→            const probe = await fetch(url, { method: "GET", cache: "no-store" });
   108→            if (!probe.ok) {
   109→              console.warn(`[pdf-viewer] worker candidate ${url} HTTP ${probe.status}`);
   110→              // Drain to free the connection.
   111→              try { await probe.arrayBuffer(); } catch {}
   112→              continue;
   113→            }
   114→            const ct = probe.headers.get("content-type") || "";
   115→            const len = Number(probe.headers.get("content-length") || "0");
   116→            // Drain to free the connection.
   117→            try { await probe.arrayBuffer(); } catch {}
   118→            if (len < 100_000) {
   119→              console.warn(`[pdf-viewer] worker candidate ${url} too small (${len}b)`);
   120→              continue;
   121→            }
   122→            if (!(ct.includes("javascript") || ct.includes("ecmascript") || ct === "")) {
   123→              console.warn(`[pdf-viewer] worker candidate ${url} wrong content-type: ${ct}`);
   124→              continue;
   125→            }
   126→
   127→            // Try to spawn a module worker.
   128→            // This may throw:
   129→            //   - SyntaxError if the script isn't valid ESM
   130→            //   - SecurityError if cross-origin without CORS
   131→            //   - NetworkError if the URL is unreachable
   132→            // We catch and fall through to the next candidate.
   133→            try {
   134→              const worker = new Worker(url, { type: "module" });
   135→              // Verify the worker is alive by sending a ping. pdf.js's
   136→              // worker will respond to "ready" but we can't easily test
   137→              // that without pdf.js. Instead we just trust the
   138→              // constructor succeeded.
   139→              lib.GlobalWorkerOptions.workerPort = worker;
   140→              workerReady = true;
   141→              console.log(`[pdf-viewer] worker ready via module Worker from ${url} (${len}b)`);
   142→              break;
   143→            } catch (spawnErr) {
   144→              console.warn(`[pdf-viewer] new Worker(${url}, {type:module}) failed:`, spawnErr);
   145→            }
   146→          } catch (probeErr) {
   147→            console.warn(`[pdf-viewer] probe ${url} failed:`, probeErr);
   148→          }
   149→        }
   150→
   151→        // ── Approach B: pdf.js workerSrc + Blob URL with polyfill ────
   152→        // Fetch the worker text, PREPEND the polyfill code (so the
   153→        // Worker thread also gets Math.sumPrecise / Map.getOrInsertComputed
   154→        // etc. — these don't transfer from the main thread), wrap in a
   155→        // Blob, hand the Blob URL to pdf.js.
   156→        //
   157→        // Why prepend: pdfjs v6 calls Math.sumPrecise inside the Worker
   158→        // thread, but the Worker has its own global scope. Main-thread
   159→        // polyfills don't apply. We must inject the polyfill into the
   160→        // worker source itself.
   161→        if (!workerReady) {
   162→          for (const url of sameOriginCandidates) {
   163→            if (workerReady) break;
   164→            try {
   165→              const res = await fetch(url, { cache: "no-store" });
   166→              if (!res.ok) continue;
   167→              const text = await res.text();
   168→              if (text.length < 100_000) continue;
   169→
   170→              // Fetch the polyfill source (synchronously loaded by
   171→              // <script src="/polyfills.js"> in <head>). We re-fetch it
   172→              // here as text so we can prepend it to the worker source.
   173→              let polyfillSource = "";
   174→              try {
   175→                const pfRes = await fetch("/polyfills.js", { cache: "no-store" });
   176→                if (pfRes.ok) {
   177→                  polyfillSource = await pfRes.text();
   178→                }
   179→              } catch (pfErr) {
   180→                console.warn("[pdf-viewer] polyfill fetch for worker failed:", pfErr);
   181→              }
   182→
   183→              const combined =
   184→                polyfillSource +
   185→                "\n;\n// ─── worker code below ───\n" +
   186→                text;
   187→              const blob = new Blob([combined], { type: "application/javascript" });
   188→              lib.GlobalWorkerOptions.workerSrc = URL.createObjectURL(blob);
   189→              workerReady = true;
   190→              console.log(
   191→                `[pdf-viewer] worker ready via Blob URL from ${url} (${text.length}b worker + ${polyfillSource.length}b polyfill)`
   192→              );
   193→              break;
   194→            } catch (blobErr) {
   195→              console.warn(`[pdf-viewer] Blob URL from ${url} failed:`, blobErr);
   196→            }
   197→          }
   198→        }
   199→
   200→        // ── Approach C: CDN workerSrc ──────────────────────────────────
   201→        if (!workerReady) {
   202→          try {
   203→            const cdnUrl = `https://cdnjs.cloudflare.com/ajax/libs/pdf.js/${lib.version}/pdf.worker.min.mjs`;
   204→            lib.GlobalWorkerOptions.workerSrc = cdnUrl;
   205→            workerReady = true;
   206→            console.warn(`[pdf-viewer] falling back to CDN worker: ${cdnUrl}`);
   207→          } catch (cdnErr) {
   208→            console.warn("[pdf-viewer] CDN worker setup failed:", cdnErr);
   209→          }
   210→        }
   211→
   212→        // ── Approach D: import worker code into main thread ───────────
   213→        // This is the ultimate fallback. We dynamically import the
   214→        // worker module directly into the main bundle. The worker code
   215→        // sets `globalThis.pdfjsWorker = { WorkerMessageHandler }` as a
   216→        // side-effect, and pdf.js detects this and runs in "fake worker"
   217→        // mode — all parsing happens on the main thread (slower but
   218→        // guaranteed to work, no spawn / no network / no CORS).
   219→        //
   220→        // We do NOT set workerSrc at all in this mode; pdf.js v6+
   221→        // detects `globalThis.pdfjsWorker` is set and skips worker
   222→        // spawning entirely.
   223→        if (!workerReady) {
   224→          try {
   225→            await import("pdfjs-dist/build/pdf.worker.min.mjs");
   226→            workerReady = true;
   227→            console.warn(
   228→              "[pdf-viewer] all worker sources failed — loaded pdf.worker.min.mjs into main thread (fake worker mode, slower but works)"
   229→            );
   230→          } catch (importErr) {
   231→            console.error("[pdf-viewer] main-thread worker import failed:", importErr);
   232→            // Truly last resort: just clear workerSrc and hope pdf.js
   233→            // can figure it out.
   234→            try {
   235→              lib.GlobalWorkerOptions.workerSrc = "";
   236→            } catch {
   237→              // ignore
   238→            }
   239→          }
   240→        }
   241→
   242→        if (mounted && !cancelled) {
   243→          pdfjsLibRef.current = lib;
   244→          setLibReady(true);
   245→        }
   246→      } catch (e) {
   247→        console.error("[pdf-viewer] init crashed unexpectedly:", e);
   248→        if (!cancelled) {
   249→          setError(
   250→            `PDF 渲染初始化失败：${e instanceof Error ? e.message : String(e)}`
   251→          );
   252→        }
   253→      }
   254→    })();
   255→    return () => {
   256→      mounted = false;
   257→      cancelled = true;
   258→    };
   259→  }, []);
   260→
   261→  // Load PDF document
   262→  useEffect(() => {
   263→    if (!fileData || !libReady || !pdfjsLibRef.current) return;
   264→    let cancelled = false;
   265→    setLoading(true);
   266→    setError(null);
   267→    setPages([]);
   268→
   269→    (async () => {
   270→      try {
   271→        const lib = pdfjsLibRef.current;
   272→        // CRITICAL: pdfjs's getDocument() TRANSFERS ownership of the
   273→        // ArrayBuffer (detaches it). If we pass the original `fileData`
   274→        // and the component later re-mounts (e.g. user switches tabs and
   275→        // comes back), the second call would see a detached buffer and
   276→        // fail with "Cannot perform construct on a detached ArrayBuffer".
   277→        //
   278→        // Fix: always slice(0) to create a fresh copy each time. The cost
   279→        // is one buffer copy per load (~1-3ms for a typical PDF), which
   280→        // is negligible compared to the actual parse cost.
   281→        const buf = fileData.slice(0);
   282→        const loadingTask = lib.getDocument({
   283→          data: new Uint8Array(buf),
   284→        });
   285→        const doc = await loadingTask.promise;
   286→        if (cancelled) return;
   287→        setPdfDoc(doc);
   288→        setNumPages(doc.numPages);
   289→
   290→        // Pre-extract text content per page (for outline keyword search)
   291→        const infos: PageInfo[] = [];
   292→        for (let i = 1; i <= doc.numPages; i++) {
   293→          const page = await doc.getPage(i);
   294→          const viewport = page.getViewport({ scale: 1 });
   295→          const tc = await page.getTextContent();
   296→          infos.push({
   297→            pageNum: i,
   298→            width: viewport.width,
   299→            height: viewport.height,
   300→            textItems: tc.items
   301→              .filter((it: any) => typeof it.str === "string" && it.str.length > 0)
   302→              .map((it: any) => ({
   303→                str: it.str,
   304→                transform: it.transform,
   305→                width: it.width,
   306→                height: it.height,
   307→              })),
   308→          });
   309→          page.cleanup();
   310→        }
   311→        if (cancelled) return;
   312→        setPages(infos);
   313→      } catch (e) {
   314→        setError(e instanceof Error ? e.message : String(e));
   315→      } finally {
   316→        if (!cancelled) setLoading(false);
   317→      }
   318→    })();
   319→
   320→    return () => {
   321→      cancelled = true;
   322→    };
   323→  }, [fileData, libReady]);
   324→
   325→  // Render pages when doc / scale changes
   326→  useEffect(() => {
   327→    if (!pdfDoc) return;
   328→    let cancelled = false;
   329→
   330→    (async () => {
   331→      for (let i = 1; i <= pdfDoc.numPages; i++) {
   332→        if (cancelled) return;
   333→        const pageDiv = pagesRef.current[i - 1];
   334→        if (!pageDiv) continue;
   335→        await renderPage(pdfDoc, i, pageDiv, scale, pdfjsLibRef.current, mode);
   336→      }
   337→    })();
   338→
   339→    return () => {
   340→      cancelled = true;
   341→    };
   342→  }, [pdfDoc, scale, mode]);
   343→
   344→  // Handle highlight token (jump + flash)
   345→  useEffect(() => {
   346→    if (!highlightToken || pages.length === 0) return;
   347→    const { quote, keywords } = highlightToken;
   348→    // Find page containing the quote or any keyword
   349→    let targetPage = -1;
   350→    let matchText = quote;
   351→    for (const p of pages) {
   352→      const full = p.textItems.map((t) => t.str).join(" ");
   353→      if (quote && quote.length > 4 && full.includes(quote)) {
   354→        targetPage = p.pageNum;
   355→        matchText = quote;
   356→        break;
   357→      }
   358→      for (const kw of keywords) {
   359→        if (kw && kw.length > 1 && full.toLowerCase().includes(kw.toLowerCase())) {
   360→          targetPage = p.pageNum;
   361→          matchText = kw;
   362→          break;
   363→        }
   364→      }
   365→      if (targetPage !== -1) break;
   366→    }
   367→    if (targetPage === -1) {
   368→      // fallback: try first keyword in any page
   369→      const kw = keywords.find((k) => k && k.length > 1);
   370→      if (kw) {
   371→        for (const p of pages) {
   372→          const full = p.textItems.map((t) => t.str).join(" ");
   373→          if (full.toLowerCase().includes(kw.toLowerCase())) {
   374→            targetPage = p.pageNum;
   375→            matchText = kw;
   376→            break;
   377→          }
   378→        }
   379→      }
   380→    }
   381→    if (targetPage !== -1) {
   382→      const pageDiv = pagesRef.current[targetPage - 1];
   383→      if (pageDiv) {
   384→        pageDiv.scrollIntoView({ behavior: "smooth", block: "center" });
   385→        // Flash highlight
   386→        pageDiv.classList.add("page-flash");
   387→        setTimeout(() => pageDiv.classList.remove("page-flash"), 1800);
   388→        // Try to highlight text spans containing the keyword
   389→        const spans = pageDiv.querySelectorAll<HTMLElement>("span.pdf-text-span");
   390→        spans.forEach((sp) => {
   391→          const t = sp.textContent || "";
   392→          if (
   393→            matchText &&
   394→            t.toLowerCase().includes(matchText.toLowerCase().slice(0, 8))
   395→          ) {
   396→            sp.classList.add("pdf-text-flash");
   397→            setTimeout(() => sp.classList.remove("pdf-text-flash"), 2400);
   398→          }
   399→        });
   400→      }
   401→    }
   402→  }, [highlightToken, pages]);
   403→
   404→  // Direct page jump — for "跳到原图 p.N" button on figure cards.
   405→  // Scrolls the target page into view + flashes it. Doesn't need text matching
   406→  // because we already know the page number.
   407→  useEffect(() => {
   408→    if (!jumpToPage || pages.length === 0) return;
   409→    const target = jumpToPage.pageIndex;
   410→    if (target < 1 || target > pages.length) return;
   411→    const pageDiv = pagesRef.current[target - 1];
   412→    if (pageDiv) {
   413→      pageDiv.scrollIntoView({ behavior: "smooth", block: "center" });
   414→      pageDiv.classList.add("page-flash");
   415→      setTimeout(() => pageDiv.classList.remove("page-flash"), 1800);
   416→    }
   417→  }, [jumpToPage, pages]);
   418→
   419→  // Text selection listener
   420→  useEffect(() => {
   421→    if (mode !== "text") return;
   422→    const handler = () => {
   423→      const sel = window.getSelection();
   424→      if (sel && sel.toString().trim().length > 0) {
   425→        onTextSelect?.(sel.toString().trim());
   426→      }
   427→    };
   428→    document.addEventListener("mouseup", handler);
   429→    return () => document.removeEventListener("mouseup", handler);
   430→  }, [mode, onTextSelect]);
   431→
   432→  const zoomIn = () => setScale((s) => Math.min(s + 0.2, 3));
   433→  const zoomOut = () => setScale((s) => Math.max(s - 0.2, 0.5));
   434→
   435→  return (
   436→    <div className="flex flex-col h-full bg-muted/20">
   437→      {/* Toolbar */}
   438→      <div className="flex items-center gap-1 px-3 py-2 border-b bg-background/80 backdrop-blur-sm">
   439→        <span className="text-xs text-muted-foreground mr-2 max-w-[200px] truncate" title={fileName}>
   440→          {fileName || "未选择文件"}
   441→        </span>
   442→        <div className="flex-1" />
   443→        <Button
   444→          variant={mode === "text" ? "default" : "ghost"}
   445→          size="sm"
   446→          className="h-7 gap-1 text-xs"
   447→          onClick={() => setMode("text")}
   448→          title="文本选择模式"
   449→        >
   450→          <MousePointer2 className="h-3.5 w-3.5" />
   451→          文本
   452→        </Button>
   453→        <Button
   454→          variant={mode === "image" ? "default" : "ghost"}
   455→          size="sm"
   456→          className="h-7 gap-1 text-xs"
   457→          onClick={() => setMode("image")}
   458→          title="图片框选模式：在 PDF 上左键拖拽选择图片区域"
   459→        >
   460→          <Square className="h-3.5 w-3.5" />
   461→          框选
   462→        </Button>
   463→        <div className="w-px h-5 bg-border mx-1" />
   464→        <Button variant="ghost" size="sm" className="h-7 w-7 p-0" onClick={zoomOut}>
   465→          <ZoomOut className="h-3.5 w-3.5" />
   466→        </Button>
   467→        <span className="text-xs text-muted-foreground w-10 text-center">
   468→          {Math.round(scale * 100)}%
   469→        </span>
   470→        <Button variant="ghost" size="sm" className="h-7 w-7 p-0" onClick={zoomIn}>
   471→          <ZoomIn className="h-3.5 w-3.5" />
   472→        </Button>
   473→      </div>
   474→
   475→      {/* Mode hint */}
   476→      {mode === "image" && fileData && (
   477→        <div className="px-3 py-1.5 bg-amber-50 border-b border-amber-200 text-xs text-amber-700 dark:bg-amber-950/40 dark:text-amber-300">
   478→          框选模式：在 PDF 页面上按住左键拖拽选择图片区域，松开后图片将自动添加到提问框
   479→        </div>
   480→      )}
   481→
   482→      {/* PDF area */}
   483→      <div
   484→        ref={containerRef}
   485→        className="flex-1 overflow-auto p-4 flex flex-col items-center gap-4"
   486→      >
   487→        {!fileData && (
   488→          <div className="flex-1 flex flex-col items-center justify-center text-muted-foreground gap-2 py-20">
   489→            <FileText className="h-10 w-10 opacity-40" />
   490→            <p className="text-sm">点击上方按钮导入 PDF 文献</p>
   491→            <p className="text-xs text-muted-foreground/70">
   492→              支持 PDF 格式；导入后将自动生成 6 维度层次化大纲
   493→            </p>
   494→          </div>
   495→        )}
   496→        {loading && (
   497→          <div className="flex items-center gap-2 text-sm text-muted-foreground py-12">
   498→            <Loader2 className="h-4 w-4 animate-spin" />
   499→            正在解析 PDF…
   500→          </div>
   501→        )}
   502→        {error && (
   503→          <div className="py-8 px-4 max-w-[600px] mx-auto">
   504→            <div className="text-sm text-red-500 font-medium mb-2">
   505→              PDF 渲染失败
   506→            </div>
   507→            <div className="text-xs text-muted-foreground mb-3 leading-relaxed">
   508→              {error}
   509→            </div>
   510→            <div className="text-[11px] text-muted-foreground/80 leading-relaxed bg-muted/40 rounded px-3 py-2 border border-border/40">
   511→              <strong className="text-foreground">可能的解决方法：</strong>
   512→              <br />· 刷新页面后重新上传 PDF（worker 文件可能加载失败）
   513→              <br />· 检查网络是否能访问同源 <code className="text-[10px] bg-muted px-1 rounded">/api/pdf-worker</code> 路径
   514→              <br />· 如错误信息含 <code className="text-[10px] bg-muted px-1 rounded">toHex</code>，说明 PDF worker 版本与 pdfjs-dist 不匹配，需重新构建（<code className="text-[10px] bg-muted px-1 rounded">npm run build</code>）
   515→            </div>
   516→          </div>
   517→        )}
   518→        {pdfDoc &&
   519→          Array.from({ length: numPages }).map((_, i) => (
   520→            <div
   521→              key={i}
   522→              ref={(el) => {
   523→                if (el) pagesRef.current[i] = el;
   524→              }}
   525→              data-page={i + 1}
   526→              className="pdf-page relative bg-white shadow-md"
   527→              style={{ border: "1px solid rgba(0,0,0,0.06)" }}
   528→            >
   529→              <canvas className="block" />
   530→              <div className="pdf-text-layer absolute inset-0" />
   531→              {mode === "image" && (
   532→                <ImageSelectOverlay
   533→                  pageNumber={i + 1}
   534→                  pdfDoc={pdfDoc}
   535→                  scale={scale}
   536→                  onCapture={(b64) => onImageSelect?.(b64)}
   537→                />
   538→              )}
   539→            </div>
   540→          ))}
   541→      </div>
   542→    </div>
   543→  );
   544→}

   548→async function renderPage(
   549→  doc: any,
   550→  pageNum: number,
   551→  pageDiv: HTMLDivElement,
   552→  scale: number,
   553→  lib: any,
   554→  mode: "text" | "image"
   555→) {
   556→  const page = await doc.getPage(pageNum);
   557→  const viewport = page.getViewport({ scale });
   558→
   559→  const canvas = pageDiv.querySelector("canvas") as HTMLCanvasElement;
   560→  const textLayer = pageDiv.querySelector(".pdf-text-layer") as HTMLDivElement;
   561→
   562→  const ratio = window.devicePixelRatio || 1;
   563→  canvas.width = Math.floor(viewport.width * ratio);
   564→  canvas.height = Math.floor(viewport.height * ratio);
   565→  canvas.style.width = `${Math.floor(viewport.width)}px`;
   566→  canvas.style.height = `${Math.floor(viewport.height)}px`;
   567→  pageDiv.style.width = `${Math.floor(viewport.width)}px`;
   568→  pageDiv.style.height = `${Math.floor(viewport.height)}px`;
   569→
   570→  const ctx = canvas.getContext("2d")!;
   571→  ctx.setTransform(ratio, 0, 0, ratio, 0, 0);
   572→
   573→  await page.render({
   574→    canvasContext: ctx,
   575→    viewport,
   576→    canvas,
   577→  }).promise;
   578→
   579→  // Build text layer
   580→  textLayer.innerHTML = "";
   581→  textLayer.style.width = `${Math.floor(viewport.width)}px`;
   582→  textLayer.style.height = `${Math.floor(viewport.height)}px`;
   583→  if (mode === "text") {
   584→    const textContent = await page.getTextContent();
   585→    // Build using TextLayer class (pdfjs v6)
   586→    try {
   587→      const tl = new lib.TextLayer({
   588→        textContentSource: textContent,
   589→        container: textLayer,
   590→        viewport,
   591→      });
   592→      await tl.render();
   593→    } catch {
   594→      // fallback: legacy renderTextLayer
   595→      try {
   596→        await lib.renderTextLayer({
   597→          textContentSource: textContent,
   598→          container: textLayer,
   599→          viewport,
   600→          textDivs: [],
   601→        }).promise;
   602→      } catch {
   603→        // ignore
   604→      }
   605→    }
   606→    // Mark spans for flash
   607→    textLayer
   608→      .querySelectorAll("span")
   609→      .forEach((sp) => sp.classList.add("pdf-text-span"));
   610→  }
   611→  page.cleanup();
   612→}

   614→// ---------- Image select overlay ----------
   615→function ImageSelectOverlay({
   616→  pageNumber,
   617→  pdfDoc,
   618→  scale,
   619→  onCapture,
   620→}: {
   621→  pageNumber: number;
   622→  pdfDoc: any;
   623→  scale: number;
   624→  onCapture: (b64: string) => void;
   625→}) {
   626→  const overlayRef = useRef<HTMLDivElement>(null);
   627→  const [dragging, setDragging] = useState(false);
   628→  const [rect, setRect] = useState<{ x: number; y: number; w: number; h: number } | null>(null);
   629→  const startRef = useRef<{ x: number; y: number } | null>(null);
   630→
   631→  const onMouseDown = (e: React.MouseEvent) => {
   632→    if (!overlayRef.current) return;
   633→    const bounds = overlayRef.current.getBoundingClientRect();
   634→    startRef.current = {
   635→      x: e.clientX - bounds.left,
   636→      y: e.clientY - bounds.top,
   637→    };
   638→    setDragging(true);
   639→    setRect({ x: startRef.current.x, y: startRef.current.y, w: 0, h: 0 });
   640→    e.preventDefault();
   641→  };
   642→
   643→  const onMouseMove = (e: React.MouseEvent) => {
   644→    if (!dragging || !startRef.current || !overlayRef.current) return;
   645→    const bounds = overlayRef.current.getBoundingClientRect();
   646→    const cx = e.clientX - bounds.left;
   647→    const cy = e.clientY - bounds.top;
   648→    const x = Math.min(startRef.current.x, cx);
   649→    const y = Math.min(startRef.current.y, cy);
   650→    const w = Math.abs(cx - startRef.current.x);
   651→    const h = Math.abs(cy - startRef.current.y);
   652→    setRect({ x, y, w, h });
   653→  };
   654→
   655→  const onMouseUp = useCallback(async () => {
   656→    if (!dragging || !rect || !overlayRef.current) {
   657→      setDragging(false);
   658→      return;
   659→    }
   660→    setDragging(false);
   661→    if (rect.w < 10 || rect.h < 10) {
   662→      setRect(null);
   663→      return;
   664→    }
   665→
   666→    try {
   667→      const page = await pdfDoc.getPage(pageNumber);
   668→      const viewport = page.getViewport({ scale });
   669→      // Render full page to offscreen canvas at devicePixelRatio.
   670→      //
   671→      // CRITICAL: We must set the ctx transform to `ratio` so the PDF
   672→      // (which renders in logical units of viewport.width ×
   673→      // viewport.height) gets scaled up to fill the full physical
   674→      // canvas (viewport.width * ratio × viewport.height * ratio).
   675→      // Without this transform, the rendered content would only fill
   676→      // the top-left corner of the canvas at 1/ratio scale, and the
   677→      // subsequent crop (which uses ratio-scaled coordinates) would
   678→      // extract mostly blank pixels OR — when the crop region extends
   679→      // past the actually-rendered area — fall back to grabbing the
   680→      // entire visible page from the main canvas instead.
   681→      const ratio = window.devicePixelRatio || 1;
   682→      const off = document.createElement("canvas");
   683→      off.width = Math.floor(viewport.width * ratio);
   684→      off.height = Math.floor(viewport.height * ratio);
   685→      const offCtx = off.getContext("2d")!;
   686→      offCtx.setTransform(ratio, 0, 0, ratio, 0, 0);
   687→      await page.render({
   688→        canvasContext: offCtx,
   689→        viewport,
   690→        canvas: off,
   691→      }).promise;
   692→
   693→      // Crop the rectangle. rect.x/y/w/h are in CSS pixels relative to
   694→      // the overlay (which is inset-0 on pageDiv), so they match the
   695→      // PDF viewport's logical coordinate system. Multiply by ratio
   696→      // to get physical-pixel offsets into the offscreen canvas.
   697→      const sx = Math.max(0, Math.floor(rect.x * ratio));
   698→      const sy = Math.max(0, Math.floor(rect.y * ratio));
   699→      const sw = Math.min(off.width - sx, Math.floor(rect.w * ratio));
   700→      const sh = Math.min(off.height - sy, Math.floor(rect.h * ratio));
   701→      if (sw < 4 || sh < 4) {
   702→        setRect(null);
   703→        return;
   704→      }
   705→      const out = document.createElement("canvas");
   706→      out.width = sw;
   707→      out.height = sh;
   708→      const outCtx = out.getContext("2d")!;
   709→      // Reset transform on the output canvas so drawImage uses raw
   710→      // physical pixel coordinates (otherwise the ratio transform
   711→      // would scale the source image down incorrectly).
   712→      outCtx.setTransform(1, 0, 0, 1, 0, 0);
   713→      outCtx.drawImage(off, sx, sy, sw, sh, 0, 0, sw, sh);
   714→      const b64 = out.toDataURL("image/png");
   715→      onCapture(b64);
   716→    } catch (e) {
   717→      console.error("image capture error", e);
   718→    } finally {
   719→      setRect(null);
   720→    }
   721→  }, [dragging, rect, pdfDoc, pageNumber, scale, onCapture]);
   722→
   723→  useEffect(() => {
   724→    if (!dragging) return;
   725→    const up = () => {
   726→      onMouseUp();
   727→    };
   728→    window.addEventListener("mouseup", up);
   729→    return () => window.removeEventListener("mouseup", up);
   730→  }, [dragging, onMouseUp]);
   731→
   732→  return (
   733→    <div
   734→      ref={overlayRef}
   735→      className="absolute inset-0 cursor-crosshair"
   736→      onMouseDown={onMouseDown}
   737→      onMouseMove={onMouseMove}
   738→    >
   739→      {rect && (
   740→        <div
   741→          className="absolute border-2 border-blue-500 bg-blue-500/15 pointer-events-none"
   742→          style={{
   743→            left: rect.x,
   744→            top: rect.y,
   745→            width: rect.w,
   746→            height: rect.h,
   747→          }}
   748→        />
   749→      )}
   750→    </div>
   751→  );
   752→}
