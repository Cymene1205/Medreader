   402→              setFiguresStatus("done");

   404→              return; // stop polling
   405→            }
   406→            if (allHaveQuestion && !spineReady) {
   407→              setFiguresStatus("spine");
   408→            } else if (figs.length > 0 && !allHaveQuestion) {
   409→              setFiguresStatus("call-a");
   410→            } else if (pollCount > 1) {

   419→              setFiguresStatus("error");
   420→              setFiguresLoading(false);
   421→              return;
   422→            }
   423→          } catch (e) {
   424→            console.warn("[poll] figures/analyze fetch failed:", e);
   425→          }
   426→          setTimeout(poll, 4000);
   427→        };
   428→        setTimeout(poll, 3000);
   429→      }
   430→    },
   431→    []

   488→  const stageLabel: Record<UploadStage, string> = {
   489→    idle: "",
   490→    uploading: "上传中…",
   491→    parsing: "MinerU 解析中…",
   492→    analyzing: "AI 分析中…",

   494→  };
