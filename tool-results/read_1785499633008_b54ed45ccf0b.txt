    79→    79→/**
    80→    80→ * Strip MinerU's over-escaped markdown punctuation + common parsing artifacts.
    81→    81→ *
    82→    82→ * MinerU outputs things like "Ehsan Vafadarnejad,\* Giuseppe Rizzo,\* ..."
    83→    83→ * The backslash before * is wrong — these are author name separators, not
    84→    84→ * emphasis markers. Strip them so the text renders cleanly.
    85→    85→ *
    86→    86→ * Also fixes several MinerU bugs that produce visible "string residue" in
    87→    87→ * the reader view:
    88→    88→ *
    89→    89→ *   1. Word-internal <sup>/<sub> tags wrapping real word content.
    90→    90→ *      MinerU sometimes mis-detects superscript boundaries and produces
    91→    91→ *      garbage like:
    92→    92→ *        "A<sup>fter</sup> <sup>acute</sup> <sup>myocardial</sup> ...
    93→    93→ *         <sup>isch-</sup>emic injury ... of<sub>1–4</sub> emic"
    94→    94→ *      Real superscripts are citation numbers / panel labels (digits and
    95→    95→ *      short symbols). If a <sup>/<sub> contains 3+ word-chars in a row,
    96→    96→ *      it's a parsing error → strip the tags, keep the inner text.
    97→    97→ *
    98→    98→ *   2. Inline LaTeX math like "$\mathsf{Ly6C}$" or "$\text{...}$" showing
    99→    99→ *      as raw text. ReactMarkdown doesn't render LaTeX by default; the raw
   100→   100→ *      `\mathsf{...}` and surrounding $ leak through. We extract the
   101→   101→ *      inner text and strip the LaTeX commands so users see "Ly6C" instead.
   102→   102→ *
   103→   103→ *   3. Stray "•" characters that MinerU sometimes inserts at line starts.
   104→   104→ *      (Only strip if at the very start of a line and not part of a
   105→   105→ *      legitimate bullet list.)
   106→   106→ */
   107→   107→function cleanMinerUText(s: string): string {
   108→   108→  if (!s) return "";
   109→   109→  let out = s;
   110→   110→
   111→   111→  // ── Fix 1: strip <sup>/<sub> tags whose content is real word text ──────
   112→   112→  // A "real" superscript is mostly digits / symbols (e.g. "1–4", "hi", "+",
   113→   113→  // "−"). If the inner text contains 3+ consecutive Latin/CJK letters, it's
   114→   114→  // almost certainly a parsing error where MinerU wrapped a real word.
   115→   115→  // We strip the tags but KEEP the inner text so the word is readable.
   116→   116→  out = out.replace(
   117→   117→    /<(sup|sub)>([^<]*)<\/\1>/gi,
   118→   118→    (fullMatch, _tag: string, inner: string) => {
   119→   119→      const t = inner.trim();
   120→   120→      if (!t) return "";
   121→   121→      // 3+ consecutive letters (Latin or CJK) inside a sup/sub = parsing error
   122→   122→      const hasLongWord = /[A-Za-z\u4e00-\u9fa5]{3,}/.test(t);
   123→   123→      // If it's mostly letters (ratio > 0.5), also treat as word content
   124→   124→      const letterCount = (t.match(/[A-Za-z]/g) || []).length;
   125→   125→      const isWordLike = t.length > 0 && letterCount / t.length > 0.5;
   126→   126→      if (hasLongWord || isWordLike) {
   127→   127→        // Strip the tags, keep the text — but preserve a leading space if
   128→   128→        // the original had one (so words don't merge).
   129→   129→        return inner;
   130→   130→      }
   131→   131→      // Otherwise it's a legit superscript (citation number, +/-, etc.)
   132→   132→      // — keep the tag intact so ReactMarkdown renders it correctly.
   133→   133→      return fullMatch;
   134→   134→    }
   135→   135→  );
   136→   136→
   137→   137→  // ── Fix 2: convert inline LaTeX math `$\cmd{...}$` to plain text ───────
   138→   138→  // Examples:
   139→   139→  //   "$\mathsf{Ly6C}$"            → "Ly6C"
   140→   140→  //   "$\text{mean} \pm SD$"       → "mean ± SD"
   141→   141→  //   "$\mathrm{CD45}^{+}$"        → "CD45+"
   142→   142→  //   "$p < 0.05$"                 → "p < 0.05"
   143→   143→  // Strategy: for each `$...$` pair, strip LaTeX commands (\mathsf, \text,
   144→   144→  // \mathrm, \frac, etc.) and braces, but keep alphanumerics + operators.
   145→   145→  out = out.replace(/\$([^$]+)\$/g, (_m, math: string) => {
   146→   146→    // Remove LaTeX commands like \mathsf, \text, \mathrm, \frac, \left, \right
   147→   147→    let cleaned = math
   148→   148→      .replace(/\\[a-zA-Z]+/g, " ")
   149→   149→      // Remove braces
   150→   150→      .replace(/[{}]/g, "")
   151→   151→      // Replace multiple spaces with single
   152→   152→      .replace(/\s+/g, " ")
   153→   153→      .trim();
   154→   154→    // If the cleaned version is just a few symbols/letters, return as-is
   155→   155→    if (cleaned.length === 0) return "";
   156→   156→    return cleaned;
   157→   157→  });
   158→   158→
   159→   159→  // ── Original fixes (MinerU escape cleanup) ─────────────────────────────
   160→   160→  // \*  → * (MinerU over-escapes asterisks used as author separators)
   161→   161→  out = out.replace(/\\\*/g, "*");
   162→   162→  // \_  → _
   163→   163→  out = out.replace(/\\_/g, "_");
   164→   164→  // \# at start of line — keep as heading marker if intended; but if escaped,
   165→   165→  // MinerU sometimes escapes inappropriately. Only unescape mid-line.
   166→   166→  out = out.replace(/(?!^)\\#/g, "#");
   167→   167→
   168→   168→  // ── Fix 3: stray leading bullet characters MinerU sometimes inserts ────
   169→   169→  // Only strip "• " at the very start of a line (legit bullets are inside
   170→   170→  // <ul><li> which ReactMarkdown handles separately).
   171→   171→  out = out.replace(/^•\s+/gm, "");
   172→   172→
   173→   173→  return out;
   174→   174→}

   214→   214→  // Each entry carries the original block index so clicking it can scroll
   215→   215→  // the reader to that exact block via blockRefs.
   216→   216→  //
   217→   217→  // ⚠️ Re-levelling: MinerU's text_level is unreliable for scientific PDFs
   218→   218→  //    (e.g. it often marks every Results subheading as level 1, making the
   219→   219→  //    navigator look like a flat list). We re-derive a 2-level hierarchy:
   220→   220→  //      - H1 = top-level structural sections (Introduction / Results /
   221→   221→  //        Discussion / Methods / References / Acknowledgements / ...)
   222→   222→  //      - H2 = sub-headings under an H1
   223→   223→  //    level >= 3 is collapsed into H2 (kept so the user can still click
   224→   224→  //    deep subsections, but rendered identically to H2 to avoid the
   225→   225→  //    "everything is one level" flat look).
   226→   226→  const headings = useMemo(() => {
   227→   227→    if (!blocks) return [];

   229→   229→      .map((b, i) => ({ block: b, idx: i }))
   230→   230→      .filter(({ block }) => {
   231→   231→        const t = (block.type || "").toLowerCase();
   232→   232→        return t === "text" && typeof block.text_level === "number" && block.text_level >= 1;
   233→   233→      })
   234→   234→      .map(({ block, idx }) => ({
   235→   235→        idx,
   236→   236→        origLevel: block.text_level || 1,
   237→   237→        level: 1, // placeholder — re-derived below
   238→   238→        text: (block.text || "").trim(),
   239→   239→      }))
   240→   240→      .filter((h) => h.text.length > 0);
   241→   241→
   242→   242→    if (raw.length === 0) return [];
   243→   243→
   244→   244→    // Re-derive H1/H2 based on the MINIMUM level seen in the document.
   245→   245→    // Whatever the lowest level MinerU emitted is, treat it as H1.
   246→   246→    // Everything strictly greater becomes H2.
   247→   247→    const minLevel = raw.reduce((m, h) => Math.min(m, h.origLevel), raw[0].origLevel);
   248→   248→    for (const h of raw) {
   249→   249→      h.level = h.origLevel === minLevel ? 1 : 2;
   250→   250→    }
   251→   251→
   252→   252→    // Heuristic: if the document has TOO FEW H1s (≤ 1), demote some H2s

   254→   254→    // papers where MinerU put the article title at level 1 and everything
   255→   255→    // else at level 2 — we want at least 3-4 H1 sections to navigate.
   256→   256→    const h1Count = raw.filter((h) => h.level === 1).length;
   257→   257→    if (h1Count <= 1 && raw.some((h) => h.level === 2)) {
   258→   258→      const h2Levels = raw
   259→   259→        .filter((h) => h.level === 2)
   260→   260→        .map((h) => h.origLevel)
   261→   261→        .sort((a, b) => a - b);
   262→   262→      const secondMin = h2Levels[0];
   263→   263→      for (const h of raw) {
   264→   264→        if (h.origLevel <= secondMin) h.level = 1;
   265→   265→        else h.level = 2;
   266→   266→      }
   267→   267→    }
   268→   268→
   269→   269→    return raw;
   270→   270→  }, [blocks]);
   271→   271→

   275→   275→    return blocks
   276→   276→      .map((b, i) => ({ ...b, _idx: i }))
   277→   277→      .filter((b) => {
   278→   278→        const t = (b.type || "").toLowerCase();
   279→   279→        return t !== "page_number" && t !== "header" && t !== "footer" && t !== "aside_text";
   280→   280→      });
   281→   281→  }, [blocks]);
   282→   282→
   283→   283→  // Image URL helper: maps MinerU's images/xxx.jpg to /api/paper-images?...
   284→   284→  const imageUrl = useCallback(
   285→   285→    (imgPath?: string): string | null => {
   286→   286→      if (!imgPath || !imagesDir) return null;
   287→   287→      const name = imgPath.split("/").pop() || "";
   288→   288→      return `/api/paper-images?dir=${encodeURIComponent(imagesDir)}&name=${encodeURIComponent(name)}`;
   289→   289→    },
   290→   290→    [imagesDir]
   291→   291→  );
   292→   292→
   293→   293→  // Imperative handle for outline clicks.
   294→   294→  useImperativeHandle(
   295→   295→    ref,
   296→   296→    () => ({
   297→   297→      scrollToText: (quote: string, keywords: string[]) => {
   298→   298→        if (!blocks || blocks.length === 0) return;
   299→   299→        const idx = findBlockIndex(blocks, quote, keywords);
   300→   300→        if (idx < 0) return;
   301→   301→        const el = blockRefs.current[idx];
   302→   302→        if (el) {
   303→   303→          el.scrollIntoView({ behavior: "smooth", block: "center" });
   304→   304→          el.classList.add("block-flash");
   305→   305→          setTimeout(() => el.classList.remove("block-flash"), 2200);
   306→   306→          setActiveIdx(idx);
   307→   307→        }
   308→   308→      },
   309→   309→    }),
   310→   310→    [blocks]
   311→   311→  );
   312→   312→
   313→   313→  // Jump to a heading by block index — used by the side drawer.
   314→   314→  const jumpToHeading = useCallback((idx: number) => {
   315→   315→    const el = blockRefs.current[idx];
   316→   316→    if (el) {
   317→   317→      el.scrollIntoView({ behavior: "smooth", block: "start" });
   318→   318→      el.classList.add("block-flash");
   319→   319→      setTimeout(() => el.classList.remove("block-flash"), 2200);
   320→   320→      setActiveIdx(idx);
   321→   321→      setActiveHeadingIdx(idx);
   322→   322→    }
   323→   323→  }, []);
   324→   324→
   325→   325→  // React to highlightToken (from outline click) — scroll & flash.
   326→   326→  useEffect(() => {
   327→   327→    if (!highlightToken || !blocks || blocks.length === 0) return;
   328→   328→    const { quote, keywords } = highlightToken;
   329→   329→    const idx = findBlockIndex(blocks, quote, keywords);
   330→   330→    if (idx < 0) return;
   331→   331→    const el = blockRefs.current[idx];
   332→   332→    if (el) {
   333→   333→      el.scrollIntoView({ behavior: "smooth", block: "center" });
   334→   334→      el.classList.add("block-flash");
   335→   335→      setTimeout(() => el.classList.remove("block-flash"), 2200);

   337→   337→    }
   338→   338→  }, [highlightToken, blocks]);
   339→   339→
   340→   340→  // Compute search matches whenever query or blocks change

   342→   342→    if (!searchQuery.trim() || !blocks) {
   343→   343→      setSearchMatches([]);
   344→   344→      setSearchCursor(0);

   346→   346→    }
   347→   347→    const q = searchQuery.toLowerCase();

   376→   376→    const idx = searchMatches[next];
   377→   377→    const el = blockRefs.current[idx];
   378→   378→    if (el) {
   379→   379→      el.scrollIntoView({ behavior: "smooth", block: "center" });
   380→   380→      el.classList.add("block-flash");
   381→   381→      setTimeout(() => el.classList.remove("block-flash"), 1500);
   382→   382→      setActiveIdx(idx);
   383→   383→    }
   384→   384→  };
   385→   385→
   386→   386→  // Keyboard shortcut: Ctrl+F / Cmd+F to open search, Esc to close
   387→   387→  useEffect(() => {
   388→   388→    const handler = (e: KeyboardEvent) => {
   389→   389→      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === "f") {
   390→   390→        e.preventDefault();

   392→   392→      } else if (e.key === "Escape") {

   394→   394→        setSearchQuery("");
   395→   395→      }
   396→   396→    };
   397→   397→    window.addEventListener("keydown", handler);
   398→   398→    return () => window.removeEventListener("keydown", handler);
   399→   399→  }, []);
   400→   400→
   401→   401→  return (
   402→   402→    <div className="flex flex-col h-full bg-muted/20">
   403→   403→      {/* Status bar */}
   404→   404→      {loading && (
   405→   405→        <div className="px-3 py-2 bg-blue-50 border-b border-blue-200 text-xs text-blue-700 dark:bg-blue-950/30 dark:text-blue-300 flex items-center gap-2">

   409→   409→      )}
   410→   410→
   411→   411→      {/* In-reader search bar */}

   420→   420→            autoFocus
   421→   421→          />
   422→   422→          <span className="text-[11px] text-muted-foreground whitespace-nowrap">
   423→   423→            {searchMatches.length > 0
   424→   424→              ? `${searchCursor + 1}/${searchMatches.length}`
   425→   425→              : searchQuery
   426→   426→              ? "无匹配"
   427→   427→              : "—"}
   428→   428→          </span>
   429→   429→          <Button
   430→   430→            variant="ghost"
   431→   431→            size="sm"
   432→   432→            className="h-7 w-7 p-0"
   433→   433→            onClick={() => gotoMatch(-1)}
   434→   434→            disabled={searchMatches.length === 0}
   435→   435→            title="上一个"
   436→   436→          >

   438→   438→          </Button>

   695→   695→function BlockView({
   696→   696→  block,
   697→   697→  index,
   698→   698→  isActive,
   699→   699→  isSearchHit,
   700→   700→  imageUrl,
   701→   701→  onParagraphClick,
   702→   702→  onActivate,
   703→   703→  blockRef,
   704→   704→  onImageClick,
   705→   705→}: {
   706→   706→  block: MinerUBlock & { _idx: number };
   707→   707→  index: number;
   708→   708→  isActive: boolean;
   709→   709→  isSearchHit: boolean;
   710→   710→  imageUrl: string | null;
   711→   711→  onParagraphClick?: (text: string, idx: number) => void;
   712→   712→  onActivate?: (idx: number) => void;

   719→   719→  const level = block.text_level;
   720→   720→
   721→   721→  // Headings

   723→   723→    const sizeCls =
   724→   724→      level === 1
   725→   725→        ? "text-2xl"
   726→   726→        : level === 2
   727→   727→        ? "text-xl"
   728→   728→        : level === 3
   729→   729→        ? "text-lg"
   730→   730→        : "text-base";
   731→   731→    return (
   732→   732→      <div ref={blockRef} data-block-idx={index} data-page={block.page_idx} className="scroll-mt-4">
   733→   733→        <div
   734→   734→          className={cn(
   735→   735→            "font-bold mt-5 mb-2 leading-tight",
   736→   736→            sizeCls,
   737→   737→            isActive && "text-primary"

   743→   743→          <div className="text-[10px] text-muted-foreground/60 -mt-1 mb-1">

   752→   752→  if (type === "table") {

   754→   754→      <div
   755→   755→        ref={blockRef}
   756→   756→        data-block-idx={index}
   757→   757→        data-page={block.page_idx}

   763→   763→        {block.table_caption && typeof block.table_caption === "string" && (

   856→   856→        data-block-idx={index}

   891→   891→          <span className="absolute left-0 top-0 bottom-0 w-[3px] bg-amber-500 rounded-r" />
   892→   892→        )}
   893→   893→        {/* Render inline markdown (bold/italic/sup/sub) so MinerU's **emphasis**

   895→   895→            ⚠️ rehypeRaw is REQUIRED here — MinerU emits raw HTML like
   896→   896→            `SiglecF<sup>hi</sup>`, `CD8<sup>+</sup>`, `IL-1β` etc. Without
   897→   897→            rehypeRaw these tags show as literal text. */}
   898→   898→        <div className="text-[14px] leading-[1.85] text-foreground/90 prose-inline-sm">
   899→   899→          <ReactMarkdown
   900→   900→            remarkPlugins={[remarkGfm]}
   901→   901→            rehypePlugins={[rehypeRaw]}
   902→   902→            disallowedElements={["p", "h1", "h2", "h3", "h4", "h5", "h6", "br", "hr", "img", "ul", "ol", "li", "blockquote", "pre"]}
   903→   903→            unwrapDisallowed
   904→   904→          >
   905→   905→            {text}
   906→   906→          </ReactMarkdown>
   907→   907→        </div>
   908→   908→      </div>
   909→   909→    );
   910→   910→  }
   911→   911→
   912→   912→  // Unknown block types — render text if present
   913→   913→  if (text && text.trim()) {
   914→   914→    return (
   915→   915→      <div
   916→   916→        ref={blockRef}
   917→   917→        data-block-idx={index}
   918→   918→        data-page={block.page_idx}
   919→   919→        className="text-[14px] leading-[1.85] text-foreground/80 scroll-mt-4"
   920→   920→      >
   921→   921→        {text}
   922→   922→      </div>
   923→   923→    );
   924→   924→  }
   925→   925→
   926→   926→  return null;

   929→   929→// ---------- helpers ----------
   930→   930→
   931→   931→/**
   932→   932→ * Find the index of the block whose text best matches the given quote or
   933→   933→ * keywords. Uses a scoring approach:
   934→   934→ *
   935→   935→ *   - exact quote substring match  → score 1000
   936→   936→ *   - quote-head (first 25 chars) substring match → score 800
   937→   937→ *   - per-keyword substring match  → +100/keyword (longer keywords weighted more)
   938→   938→ *   - prefer blocks where MORE keywords match
   939→   939→ *
   940→   940→ * Returns the highest-scoring block index, or -1 if no block scored > 0.
   941→   941→ */
   942→   942→function findBlockIndex(
   943→   943→  blocks: MinerUBlock[],
   944→   944→  quote: string,
   945→   945→  keywords: string[]
   946→   946→): number {
   947→   947→  if (!blocks || blocks.length === 0) return -1;
   948→   948→
   949→   949→  // Normalize a string for matching: lowercase, collapse whitespace.
   950→   950→  // Also strip <sup>/<sub>/<i> HTML tags so e.g. "SiglecF<sup>hi</sup>"
   951→   951→  // matches a search for "siglecfhi".
   952→   952→  const norm = (s: string) =>
   953→   953→    (s || "")
   954→   954→      .toLowerCase()
   955→   955→      .replace(/<\/?(?:sup|sub|i|b|em|strong|span)[^>]*>/gi, "")
   956→   956→      .replace(/\s+/g, " ")
   957→   957→      .trim();
   958→   958→
   959→   959→  const q = norm(quote || "");
   960→   960→  const qHead = q.slice(0, 25);
   961→   961→  const kws = (keywords || [])
   962→   962→    .filter((k) => k && k.length >= 2)
   963→   963→    .map(norm)
   964→   964→    .sort((a, b) => b.length - a.length); // longest first
   965→   965→
   966→   966→  // Extract any "Figure N" / "Fig N" / "Fig NA" references from the quote
   967→   967→  // itself — if the LLM included the figure callout in its quote, we want
   968→   968→  // blocks that mention the same figure to score higher.
   969→   969→  const figRefsInQuote = new Set<string>();
   970→   970→  const figRefRe = /\b(?:fig(?:ure|\.)?)\s*(\d+)\s*([a-z])?/gi;
   971→   971→  let m: RegExpExecArray | null;
   972→   972→  while ((m = figRefRe.exec(q)) !== null) {
   973→   973→    const num = m[1];
   974→   974→    const panel = m[2] || "";
   975→   975→    figRefsInQuote.add(`figure ${num}`);
   976→   976→    if (panel) {
   977→   977→      figRefsInQuote.add(`figure ${num}${panel}`);
   978→   978→      figRefsInQuote.add(`fig. ${num}`);
   979→   979→      figRefsInQuote.add(`fig ${num}${panel}`);
   980→   980→    }
   981→   981→  }
   982→   982→
   983→   983→  let bestIdx = -1;
   984→   984→  let bestScore = 0;
   985→   985→
   986→   986→  for (let i = 0; i < blocks.length; i++) {
   987→   987→    const t = norm(blocks[i].text || "");
   988→   988→    if (!t) continue;
   989→   989→
   990→   990→    let score = 0;
   991→   991→
   992→   992→    // 1) Exact quote match
   993→   993→    if (q.length > 4 && t.includes(q)) {
   994→   994→      score += 1000;
   995→   995→    }
   996→   996→    // 2) Quote head match (LLM often paraphrases the tail)
   997→   997→    else if (qHead.length > 6 && t.includes(qHead)) {
   998→   998→      score += 800;
   999→   999→    }
  1000→  1000→
  1001→  1001→    // 3) Keyword matches — longer keywords score more
  1002→  1002→    for (const kw of kws) {
  1003→  1003→      if (t.includes(kw)) {
  1004→  1004→        // Weight by length: 4-char keyword = 100, 10-char = 250
  1005→  1005→        score += Math.min(300, 25 + kw.length * 25);
  1006→  1006→      }
  1007→  1007→    }
  1008→  1008→
  1009→  1009→    // 4) Token overlap bonus: split quote into words, count how many appear
  1010→  1010→    if (q.length > 4) {
  1011→  1011→      const tokens = q.split(/[,\s.;:()]+/).filter((w) => w.length > 3);
  1012→  1012→      let hits = 0;
  1013→  1013→      for (const tok of tokens) {
  1014→  1014→        if (t.includes(tok)) hits++;
  1015→  1015→      }
  1016→  1016→      if (tokens.length > 0) {
  1017→  1017→        score += Math.round((hits / tokens.length) * 200);
  1018→  1018→      }
  1019→  1019→    }
  1020→  1020→
  1021→  1021→    // 5) Figure-reference bonus — if the quote mentions "Figure 1A", boost
  1022→  1022→    // blocks that also mention "Figure 1" / "Fig. 1A". This is the panel
  1023→  1023→    // jump fallback when the LLM's quote is too short or paraphrased.
  1024→  1024→    if (figRefsInQuote.size > 0) {
  1025→  1025→      for (const ref of figRefsInQuote) {
  1026→  1026→        if (t.includes(ref)) {
  1027→  1027→          score += 150;
  1028→  1028→          break;
  1029→  1029→        }
  1030→  1030→      }
  1031→  1031→    }
  1032→  1032→
  1033→  1033→    if (score > bestScore) {
  1034→  1034→      bestScore = score;
  1035→  1035→      bestIdx = i;
  1036→  1036→    }
  1037→  1037→  }
  1038→  1038→
  1039→  1039→  // Require at least some signal — avoid jumping to block 0 on no match
  1040→  1040→  return bestScore > 0 ? bestIdx : -1;
  1041→  1041→}
