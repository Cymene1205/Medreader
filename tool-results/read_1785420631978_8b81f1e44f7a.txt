   107→// ---------- Helpers ----------
   108→function formatDateShort(iso: string | null): string {
   109→  if (!iso) return "—";
   110→  const d = new Date(iso);
   111→  if (Number.isNaN(d.getTime())) return "—";
   112→  return `${d.getMonth() + 1}/${d.getDate()}`;
   113→}
   114→
   115→function formatDateTime(iso: string | null): string {
   116→  if (!iso) return "—";
   117→  const d = new Date(iso);
   118→  if (Number.isNaN(d.getTime())) return "—";
   119→  const y = d.getFullYear();
   120→  const m = String(d.getMonth() + 1).padStart(2, "0");
   121→  const day = String(d.getDate()).padStart(2, "0");
   122→  const hh = String(d.getHours()).padStart(2, "0");
   123→  const mm = String(d.getMinutes()).padStart(2, "0");
   124→  return `${y}-${m}-${day} ${hh}:${mm}`;
   125→}
   126→
   127→function datePart(iso: string | null): string {
   128→  if (!iso) return "";
   129→  // ISO strings start with YYYY-MM-DD regardless of T or space separator.
   130→  return iso.slice(0, 10);
   131→}

   619→              )}

   622→              </div>
   623→            </div>
   624→
   625→            {loading ? (
   626→              <Skeleton className="h-[300px] w-full" />

   628→              <div className="max-h-[520px] overflow-y-auto scrollbar-thin border-t">

   705→      <footer className="mt-auto border-t border-border bg-muted/30">
