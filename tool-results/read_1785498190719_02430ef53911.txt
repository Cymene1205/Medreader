   161→   161→
   162→   162→export default function OutlinePanel({
   163→   163→  outline,
   164→   164→  loading,
   165→   165→  collapsed = false,
   166→   166→  onCollapsedChange,
   167→   167→  paperId,
   168→   168→  figures = [],
   169→   169→  citations = [],
   170→   170→  onPanelChipClick,
   171→   171→  onJumpToPage,
   172→   172→  figuresStatus = "idle",
   173→   173→  uploadStage = "idle",
   174→   174→  mineruStatus = "",
   175→   175→}: Props) {
   176→   176→  const [retrying, setRetrying] = useState<string | null>(null);
   177→   177→
   178→   178→  // ── Per-section fold state ────────────────────────────────────────────
   179→   179→  // User request: "开始的时候就展开成这样就可以了,不用把所有的内容都展开"
   180→   180→  // → only ONE section expanded by default (问题与背景), the other three
   181→   181→  //   collapsed. User can click any header to expand it.
   182→   182→  // The state is keyed by SectionKey. We initialize lazily so a freshly
   183→   183→  // loaded outline starts in the right fold state regardless of when its
   184→   184→  // content arrives.
   185→   185→  const [openSections, setOpenSections] = useState<Record<SectionKey, boolean>>({
   186→   186→    questionBackground: true,   // 默认展开第一个
   187→   187→    argumentSpine: false,
   188→   188→    novelty: false,
   189→   189→    limitsOpportunities: false,
   190→   190→  });
   191→   191→  const toggleSection = useCallback((key: SectionKey) => {
   192→   192→    setOpenSections((prev) => ({ ...prev, [key]: !prev[key] }));
   193→   193→  }, []);
   194→   194→
   195→   195→  // 当一个全新的 outline 到达时，重置回默认折叠态（仅 questionBackground 展开）。
   196→   196→  // 否则切换论文时上一份的展开状态会残留。
   197→   197→  const outlineTitleRef = useRef<string | null>(null);
   198→   198→  useEffect(() => {
   199→   199→    if (outline?.title && outlineTitleRef.current !== outline.title) {
   200→   200→      outlineTitleRef.current = outline.title;
   201→   201→      setOpenSections({
   202→   202→        questionBackground: true,
   203→   203→        argumentSpine: false,
   204→   204→        novelty: false,
   205→   205→        limitsOpportunities: false,
   206→   206→      });
   207→   207→    }
   208→   208→  }, [outline?.title]);
   209→   209→
   210→   210→  const fs = (px: number) => `${px}px`;
   211→   211→
   212→   212→  // Retry a failed part
   213→   213→  const retryPart = useCallback(
   214→   214→    async (part: string) => {
   215→   215→      if (!paperId) return;
   216→   216→      setRetrying(part);
   217→   217→      try {
   218→   218→        const res = await fetch(`/api/analyze?part=${part}`, {
   219→   219→          method: "POST",
   220→   220→          headers: { "Content-Type": "application/json" },
   221→   221→          body: JSON.stringify({ paperId }),
   222→   222→        });
   223→   223→        if (!res.ok) {
   224→   224→          const d = await res.json().catch(() => ({}));
   225→   225→          throw new Error(d.error || `HTTP ${res.status}`);
   226→   226→        }
   227→   227→        // Trigger parent refresh — parent will re-fetch analysisJson
   228→   228→        window.dispatchEvent(new CustomEvent("medreader:analysis-updated"));
   229→   229→      } catch (e) {
   230→   230→        console.warn(`[outline-panel] retry ${part} failed:`, e);
   231→   231→      } finally {
   232→   232→        setRetrying(null);
   233→   233→      }
   234→   234→    },
   235→   235→    [paperId]
   236→   236→  );
   237→   237→
   238→   238→  // ── Header (always rendered) ────────────────────────────────────────────
   239→   239→  // Determine whether we're in the upload + parse + analyze pipeline.
   240→   240→  // User request: "刚开始现在不是有minerU解析框我很喜欢,可以吧agent分析
   241→   241→  // 的那个进程也放在那里显示一致一点,进度条拉长一点儿把agent分析也放进去"
   242→   242→  // → unify the MinerU loading indicator and the Agent analysis indicator
   243→   243→  //   into a single 4-step progress bar (上传 PDF → MinerU 解析 → 图表提取
   244→   244→  //   → 智能分析). Show the same component for both phases so the user sees
   245→   245→  //   one continuous progress flow on the left side.
   246→   246→  const isMineruLoading =
   247→   247→    !outline && (uploadStage === "uploading" || uploadStage === "parsing");
   248→   248→  // True while Stage 1 LLM analysis is running (after MinerU finished).
   249→   249→  const isAnalyzing =
   250→   250→    loading && !outline && uploadStage === "analyzing";
   251→   251→  // True once Stage 2 (figure extraction + Call A + spine) is in flight
   252→   252→  // AND the outline isn't ready yet — adds a "图表提取" sub-step to the bar.
   253→   253→  const isExtractingFigures =
   254→   254→    !outline && (figuresStatus === "extracting" || figuresStatus === "call-a");
   255→   255→
   256→   256→  return (
   257→   257→    <div className={cn("flex flex-col bg-card", collapsed ? "h-auto" : "h-full")}>
   258→   258→      {/* Panel header — always visible, click to collapse/expand the whole panel.
   259→   259→          Restored per user request: "我还是喜欢之前的那种排版方式,右上方展开栏".
   260→   260→          The outer `collapsed` prop controls the panel-level collapse (whole
   261→   261→          panel hidden except header); individual sections inside have their
   262→   262→          own per-section fold state (see openSections above). */}
   263→   263→      <button
   264→   264→        type="button"
   265→   265→        onClick={() => onCollapsedChange?.(!collapsed)}
   266→   266→        className={cn(
   267→   267→          "w-full px-3 py-2 flex items-center gap-2 text-left flex-shrink-0 transition-colors",
   268→   268→          "hover:bg-slate-100/60 dark:hover:bg-slate-800/40",
   269→   269→          "border-b border-slate-200/70 dark:border-slate-800/60",
   270→   270→          "bg-gradient-to-r from-slate-50/80 to-slate-100/40 dark:from-slate-900/60 dark:to-slate-900/30"
   271→   271→        )}
   272→   272→        title={collapsed ? "展开全文框架" : "折叠全文框架"}
   273→   273→        aria-label={collapsed ? "展开全文框架" : "折叠全文框架"}
   274→   274→        aria-expanded={!collapsed}
   275→   275→      >
   276→   276→        <LayoutGrid className="h-3.5 w-3.5 text-slate-600 dark:text-slate-400 flex-shrink-0" />
   277→   277→        <span className="text-[12px] font-semibold flex-1 text-slate-700 dark:text-slate-300">
   278→   278→          全文框架
   279→   279→        </span>
   280→   280→        {outline && (
   281→   281→          <Badge
   282→   282→            variant="secondary"
   283→   283→            className="text-[10px] h-4 px-1.5 bg-slate-200/70 text-slate-700 dark:bg-slate-800/60 dark:text-slate-300"
   284→   284→          >
   285→   285→            4 层
   286→   286→          </Badge>
   287→   287→        )}
   288→   288→        {/* 右上方展开/折叠按钮 — restores the previous layout's panel-level toggle */}
   289→   289→        <ChevronRight
   290→   290→          className={cn(
   291→   291→            "h-3.5 w-3.5 text-muted-foreground transition-transform flex-shrink-0",
   292→   292→            !collapsed && "rotate-90"
   293→   293→          )}
   294→   294→        />
   295→   295→      </button>
   296→   296→
   297→   297→      {!collapsed && (
   298→   298→        <div className="flex-1 min-h-0 overflow-y-auto scrollbar-thin">
   299→   299→          <div className="p-2">
   300→   300→            {/* Unified loading indicator — shown for the ENTIRE upload →
   301→   301→                MinerU parse → figure extraction → Agent analysis pipeline.
   302→   302→                User request: unify so the user sees one continuous progress
   303→   303→                bar on the left side rather than two different loading UIs. */}
   304→   304→            {(isMineruLoading || isAnalyzing || isExtractingFigures) && (
   305→   305→              <PipelineLoadingIndicator
   306→   306→                uploadStage={uploadStage}
   307→   307→                figuresStatus={figuresStatus}
   308→   308→                figuresCount={figures.length}
   309→   309→                statusMessage={mineruStatus}
   310→   310→              />
   311→   311→            )}
   312→   312→
   313→   313→            {/* 完全空闲：还没传 PDF */}
   314→   314→            {!loading && !outline && uploadStage === "idle" && (
   315→   315→              <div className="flex flex-col items-center justify-center py-12 text-muted-foreground/70 gap-2 px-4 text-center">
   316→   316→                <FileSearch className="h-6 w-6 opacity-40" />
   317→   317→                <p className="text-xs">导入 PDF 后，Agent 将自动生成分析</p>
   318→   318→                <p className="text-[10px]">
   319→   319→                  4 层结构：问题与背景 · 论证主线 · 创新性 · 局限与机会
   320→   320→                </p>
   321→   321→              </div>
   322→   322→            )}
   323→   323→
   324→   324→            {outline && (
   325→   325→              <div className="space-y-1.5">
   326→   326→                {SECTIONS.map((sec) => {
   327→   327→                  const part = outline[sec.key];
   328→   328→                  // Per-section fold state — defaults set above
   329→   329→                  // (argumentSpine collapsed, others expanded).
   330→   330→                  const isOpen = openSections[sec.key];
   331→   331→                  const isFailed = outline.failedParts?.includes(sec.key);
   332→   332→                  const isRetrying = retrying === sec.key;
   333→   333→
   334→   334→                  // Determine if section has content
   335→   335→                  const hasContent = (() => {
   336→   336→                    if (sec.key === "argumentSpine") {
   337→   337→                      return outline.argumentSpine?.summary || figures.length > 0;
   338→   338→                    }
   339→   339→                    if (sec.key === "limitsOpportunities") {
   340→   340→                      return outline.limitsOpportunities?.summary;
   341→   341→                    }
   342→   342→                    return part && (part as any).summary;
   343→   343→                  })();
   344→   344→
   345→   345→                  return (
   346→   346→                    <div
   347→   347→                      key={sec.key}
   348→   348→                      className={cn(
   349→   349→                        "rounded-md border bg-card shadow-sm transition-all"
   350→   350→                      )}
   351→   351→                      style={{
   352→   352→                        borderLeft: `3px solid ${sec.color}`,
   353→   353→                        borderColor: sec.border,
   354→   354→                      }}
   355→   355→                    >
   356→   356→                      {/* Header — clickable to fold/unfold the section.
   357→   357→                          Main sections (Q&A, Novelty, Limits) start expanded;
   358→   358→                          ArgumentSpine starts collapsed (per user request:
   359→   359→                          "大标题默认展开, 论证主线默认缩起来"). */}
   360→   360→                      <button
   361→   361→                        type="button"
   362→   362→                        onClick={() => toggleSection(sec.key)}
   363→   363→                        aria-expanded={isOpen}
   364→   364→                        className={cn(
   365→   365→                          "w-full text-left px-2.5 py-2 flex items-start gap-2",
   366→   366→                          "rounded-t-md transition-colors hover:bg-black/[0.03] dark:hover:bg-white/[0.04]",
   367→   367→                          "focus:outline-none focus-visible:ring-1 focus-visible:ring-inset focus-visible:ring-primary/40",
   368→   368→                          !isOpen && "rounded-b-md"
   369→   369→                        )}
   370→   370→                        style={{ background: sec.soft }}
   371→   371→                      >
   372→   372→                        {/* Chevron fold indicator */}
   373→   373→                        <span className="flex-shrink-0 w-3 mt-0.5 flex items-center justify-center">
   374→   374→                          {isOpen ? (
   375→   375→                            <ChevronDown
   376→   376→                              className="h-3 w-3"
   377→   377→                              style={{ color: sec.color }}
   378→   378→                            />
   379→   379→                          ) : (
   380→   380→                            <ChevronRight
   381→   381→                              className="h-3 w-3"
   382→   382→                              style={{ color: sec.color }}
   383→   383→                            />
   384→   384→                          )}
   385→   385→                        </span>
   386→   386→                        <span
   387→   387→                          className="flex-shrink-0 w-5 h-5 rounded text-[10px] font-bold text-white flex items-center justify-center mt-0.5"
   388→   388→                          style={{ background: sec.color }}
   389→   389→                        >
   390→   390→                          {sec.index}
   391→   391→                        </span>
   392→   392→                        <div className="flex-1 min-w-0">
   393→   393→                          <div
   394→   394→                            className="font-medium leading-snug"
   395→   395→                            style={{ fontSize: fs(13), color: sec.color }}
   396→   396→                          >
   397→   397→                            {sec.title}
   398→   398→                          </div>
   399→   399→                          {hasContent && (part as any)?.summary && (
   400→   400→                            <div
   401→   401→                              className={cn(
   402→   402→                                "text-muted-foreground mt-0.5 whitespace-pre-wrap break-words",
   403→   403→                                // 旧版折叠态会 line-clamp-2 截断 summary，
   404→   404→                                // 用户反馈「希望展示完整」——现在无论开合都完整展示。
   405→   405→                                "line-clamp-none"
   406→   406→                              )}
   407→   407→                              style={{ fontSize: fs(11) }}
   408→   408→                            >
   409→   409→                              {stripMarkdownInline((part as any).summary)}
   410→   410→                            </div>
   411→   411→                          )}
   412→   412→                          {!hasContent && !isFailed && sec.key !== "argumentSpine" && (
   413→   413→                            <div
   414→   414→                              className="text-muted-foreground/60 mt-0.5 italic"
   415→   415→                              style={{ fontSize: fs(11) }}
   416→   416→                            >
   417→   417→                              正在生成…
   418→   418→                            </div>
   419→   419→                          )}
   420→   420→                          {!hasContent && !isFailed && sec.key === "argumentSpine" && (
   421→   421→                            <ArgumentSpineProgress
   422→   422→                              figuresStatus={figuresStatus}
   423→   423→                              figuresCount={figures.length}
   424→   424→                              hasSpine={!!outline.argumentSpine?.summary}
   425→   425→                            />
