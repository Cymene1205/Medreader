   107→   107→// ---------- Helpers ----------
   108→   108→function formatDateShort(iso: string | null): string {
   109→   109→  if (!iso) return "—";
   110→   110→  const d = new Date(iso);
   111→   111→  if (Number.isNaN(d.getTime())) return "—";
   112→   112→  return `${d.getMonth() + 1}/${d.getDate()}`;
   113→   113→}
   114→   114→
   115→   115→function formatDateTime(iso: string | null): string {
   116→   116→  if (!iso) return "—";
   117→   117→  const d = new Date(iso);
   118→   118→  if (Number.isNaN(d.getTime())) return "—";
   119→   119→  const y = d.getFullYear();
   120→   120→  const m = String(d.getMonth() + 1).padStart(2, "0");
   121→   121→  const day = String(d.getDate()).padStart(2, "0");
   122→   122→  const hh = String(d.getHours()).padStart(2, "0");
   123→   123→  const mm = String(d.getMinutes()).padStart(2, "0");
   124→   124→  return `${y}-${m}-${day} ${hh}:${mm}`;
   125→   125→}
   126→   126→
   127→   127→function datePart(iso: string | null): string {
   128→   128→  if (!iso) return "";
   129→   129→  // ISO strings start with YYYY-MM-DD regardless of T or space separator.
   130→   130→  return iso.slice(0, 10);
   131→   131→}
