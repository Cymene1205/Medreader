    79→/**
    80→ * Strip MinerU's over-escaped markdown punctuation + common parsing artifacts.
    81→ *
    82→ * MinerU outputs things like "Ehsan Vafadarnejad,\* Giuseppe Rizzo,\* ..."
    83→ * The backslash before * is wrong — these are author name separators, not
    84→ * emphasis markers. Strip them so the text renders cleanly.
    85→ *
    86→ * Also fixes several MinerU bugs that produce visible "string residue" in
    87→ * the reader view:
    88→ *
    89→ *   1. Word-internal <sup>/<sub> tags wrapping real word content.
    90→ *      MinerU sometimes mis-detects superscript boundaries and produces
    91→ *      garbage like:
    92→ *        "A<sup>fter</sup> <sup>acute</sup> <sup>myocardial</sup> ...
    93→ *         <sup>isch-</sup>emic injury ... of<sub>1–4</sub> emic"
    94→ *      Real superscripts are citation numbers / panel labels (digits and
    95→ *      short symbols). If a <sup>/<sub> contains 3+ word-chars in a row,
    96→ *      it's a parsing error → strip the tags, keep the inner text.
    97→ *
    98→ *   2. Inline LaTeX math like "$\mathsf{Ly6C}$" or "$\text{...}$" showing
    99→ *      as raw text. ReactMarkdown doesn't render LaTeX by default; the raw
   100→ *      `\mathsf{...}` and surrounding $ leak through. We extract the
   101→ *      inner text and strip the LaTeX commands so users see "Ly6C" instead.
   102→ *
   103→ *   3. Stray "•" characters that MinerU sometimes inserts at line starts.
   104→ *      (Only strip if at the very start of a line and not part of a
   105→ *      legitimate bullet list.)
   106→ */
   107→function cleanMinerUText(s: string): string {
   108→  if (!s) return "";
   109→  let out = s;
   110→
   111→  // ── Fix 1: strip <sup>/<sub> tags whose content is real word text ──────
   112→  // A "real" superscript is mostly digits / symbols (e.g. "1–4", "hi", "+",
   113→  // "−"). If the inner text contains 3+ consecutive Latin/CJK letters, it's
   114→  // almost certainly a parsing error where MinerU wrapped a real word.
   115→  // We strip the tags but KEEP the inner text so the word is readable.
   116→  out = out.replace(
   117→    /<(sup|sub)>([^<]*)<\/\1>/gi,
   118→    (fullMatch, _tag: string, inner: string) => {
   119→      const t = inner.trim();
   120→      if (!t) return "";
   121→      // 3+ consecutive letters (Latin or CJK) inside a sup/sub = parsing error
   122→      const hasLongWord = /[A-Za-z\u4e00-\u9fa5]{3,}/.test(t);
   123→      // If it's mostly letters (ratio > 0.5), also treat as word content
   124→      const letterCount = (t.match(/[A-Za-z]/g) || []).length;
   125→      const isWordLike = t.length > 0 && letterCount / t.length > 0.5;
   126→      if (hasLongWord || isWordLike) {
   127→        // Strip the tags, keep the text — but preserve a leading space if
   128→        // the original had one (so words don't merge).
   129→        return inner;
   130→      }
   131→      // Otherwise it's a legit superscript (citation number, +/-, etc.)
   132→      // — keep the tag intact so ReactMarkdown renders it correctly.
   133→      return fullMatch;
   134→    }
   135→  );
   136→
   137→  // ── Fix 2: convert inline LaTeX math `$\cmd{...}$` to plain text ───────
   138→  // Examples:
   139→  //   "$\mathsf{Ly6C}$"            → "Ly6C"
   140→  //   "$\text{mean} \pm SD$"       → "mean ± SD"
   141→  //   "$\mathrm{CD45}^{+}$"        → "CD45+"
   142→  //   "$p < 0.05$"                 → "p < 0.05"
   143→  // Strategy: for each `$...$` pair, strip LaTeX commands (\mathsf, \text,
   144→  // \mathrm, \frac, etc.) and braces, but keep alphanumerics + operators.
   145→  out = out.replace(/\$([^$]+)\$/g, (_m, math: string) => {
   146→    // Remove LaTeX commands like \mathsf, \text, \mathrm, \frac, \left, \right
   147→    let cleaned = math
   148→      .replace(/\\[a-zA-Z]+/g, " ")
   149→      // Remove braces
   150→      .replace(/[{}]/g, "")
   151→      // Replace multiple spaces with single
   152→      .replace(/\s+/g, " ")
   153→      .trim();
   154→    // If the cleaned version is just a few symbols/letters, return as-is
   155→    if (cleaned.length === 0) return "";
   156→    return cleaned;
   157→  });
   158→
   159→  // ── Original fixes (MinerU escape cleanup) ─────────────────────────────
   160→  // \*  → * (MinerU over-escapes asterisks used as author separators)
   161→  out = out.replace(/\\\*/g, "*");
   162→  // \_  → _
   163→  out = out.replace(/\\_/g, "_");
   164→  // \# at start of line — keep as heading marker if intended; but if escaped,
   165→  // MinerU sometimes escapes inappropriately. Only unescape mid-line.
   166→  out = out.replace(/(?!^)\\#/g, "#");
   167→
   168→  // ── Fix 3: stray leading bullet characters MinerU sometimes inserts ────
   169→  // Only strip "• " at the very start of a line (legit bullets are inside
   170→  // <ul><li> which ReactMarkdown handles separately).
   171→  out = out.replace(/^•\s+/gm, "");
   172→
   173→  return out;
   174→}

   176→/**
   177→ * Block-level reader — renders MinerU's content_list.json as a stream of
   178→ * discrete blocks (headings, paragraphs, tables, figures, equations).
   179→ *
   180→ * - Single-click a paragraph → triggers onParagraphClick for translation.

   182→ * - In-reader search bar (Ctrl+F-like) — highlights & navigates matches.
   183→ */
   184→const BlockReader = forwardRef<BlockReaderHandle, Props>(function BlockReader(
   185→  { fallbackText, markdown, blocks, imagesDir, loading, statusMessage, onParagraphClick, highlightToken },
   186→  ref
   187→) {
   188→  const containerRef = useRef<HTMLDivElement>(null);
   189→  const blockRefs = useRef<(HTMLDivElement | null)[]>([]);
   190→  const [activeIdx, setActiveIdx] = useState<number | null>(null);
   191→
   192→  // In-reader search state
   193→  const [searchOpen, setSearchOpen] = useState(false);
   194→  const [searchQuery, setSearchQuery] = useState("");
   195→  const [searchMatches, setSearchMatches] = useState<number[]>([]);
   196→  const [searchCursor, setSearchCursor] = useState(0);
   197→
   198→  // Heading navigator side-drawer state
   199→  const [headingsOpen, setHeadingsOpen] = useState(false);
   200→  const [activeHeadingIdx, setActiveHeadingIdx] = useState<number | null>(null);
   201→
   202→  // Image preview dialog — clicking any figure opens a large preview
   203→  // with the full caption rendered below.
   204→  const [imgPreview, setImgPreview] = useState<{
   205→    imageUrl: string;
   206→    caption: string;
   207→    label?: string;
   208→  } | null>(null);
   209→
   210→  const hasBlocks = blocks && blocks.length > 0;
   211→
   212→  // Extract heading blocks (type === "text" with text_level >= 1) — used by
   213→  // the side drawer. No LLM involved, just MinerU's structural metadata.
   214→  // Each entry carries the original block index so clicking it can scroll
   215→  // the reader to that exact block via blockRefs.
   216→  //
   217→  // ⚠️ Re-levelling: MinerU's text_level is unreliable for scientific PDFs
   218→  //    (e.g. it often marks every Results subheading as level 1, making the
   219→  //    navigator look like a flat list). We re-derive a 2-level hierarchy:
   220→  //      - H1 = top-level structural sections (Introduction / Results /
   221→  //        Discussion / Methods / References / Acknowledgements / ...)
   222→  //      - H2 = sub-headings under an H1
   223→  //    level >= 3 is collapsed into H2 (kept so the user can still click
   224→  //    deep subsections, but rendered identically to H2 to avoid the
   225→  //    "everything is one level" flat look).
   226→  const headings = useMemo(() => {
   227→    if (!blocks) return [];
   228→    const raw = blocks
   229→      .map((b, i) => ({ block: b, idx: i }))
   230→      .filter(({ block }) => {
   231→        const t = (block.type || "").toLowerCase();
   232→        return t === "text" && typeof block.text_level === "number" && block.text_level >= 1;
   233→      })
   234→      .map(({ block, idx }) => ({
   235→        idx,
   236→        origLevel: block.text_level || 1,
   237→        level: 1, // placeholder — re-derived below
   238→        text: (block.text || "").trim(),
   239→      }))
   240→      .filter((h) => h.text.length > 0);
   241→
   242→    if (raw.length === 0) return [];
   243→
   244→    // Re-derive H1/H2 based on the MINIMUM level seen in the document.
   245→    // Whatever the lowest level MinerU emitted is, treat it as H1.
   246→    // Everything strictly greater becomes H2.
   247→    const minLevel = raw.reduce((m, h) => Math.min(m, h.origLevel), raw[0].origLevel);
   248→    for (const h of raw) {
   249→      h.level = h.origLevel === minLevel ? 1 : 2;
   250→    }
   251→
   252→    // Heuristic: if the document has TOO FEW H1s (≤ 1), demote some H2s
   253→    // to H1 by treating the second-lowest level as H1 too. This handles
   254→    // papers where MinerU put the article title at level 1 and everything
   255→    // else at level 2 — we want at least 3-4 H1 sections to navigate.
   256→    const h1Count = raw.filter((h) => h.level === 1).length;
   257→    if (h1Count <= 1 && raw.some((h) => h.level === 2)) {
   258→      const h2Levels = raw
   259→        .filter((h) => h.level === 2)
   260→        .map((h) => h.origLevel)
   261→        .sort((a, b) => a - b);
   262→      const secondMin = h2Levels[0];
   263→      for (const h of raw) {
   264→        if (h.origLevel <= secondMin) h.level = 1;
   265→        else h.level = 2;
   266→      }
   267→    }
   268→
   269→    return raw;
   270→  }, [blocks]);
   271→
   272→  // Filter blocks to only render the readable content (skip page_number, footer noise).
   273→  const renderBlocks = useMemo(() => {
   274→    if (!blocks) return [];
   275→    return blocks
   276→      .map((b, i) => ({ ...b, _idx: i }))
   277→      .filter((b) => {
   278→        const t = (b.type || "").toLowerCase();
   279→        return t !== "page_number" && t !== "header" && t !== "footer" && t !== "aside_text";
   280→      });
   281→  }, [blocks]);
   282→
   283→  // Image URL helper: maps MinerU's images/xxx.jpg to /api/paper-images?...
   284→  const imageUrl = useCallback(
   285→    (imgPath?: string): string | null => {
   286→      if (!imgPath || !imagesDir) return null;
   287→      const name = imgPath.split("/").pop() || "";
   288→      return `/api/paper-images?dir=${encodeURIComponent(imagesDir)}&name=${encodeURIComponent(name)}`;
   289→    },
   290→    [imagesDir]
   291→  );
   292→
   293→  // Imperative handle for outline clicks.
   294→  useImperativeHandle(
   295→    ref,
   296→    () => ({
   297→      scrollToText: (quote: string, keywords: string[]) => {
   298→        if (!blocks || blocks.length === 0) return;
   299→        const idx = findBlockIndex(blocks, quote, keywords);
   300→        if (idx < 0) return;
   301→        const el = blockRefs.current[idx];
   302→        if (el) {
   303→          el.scrollIntoView({ behavior: "smooth", block: "center" });
   304→          el.classList.add("block-flash");
   305→          setTimeout(() => el.classList.remove("block-flash"), 2200);
   306→          setActiveIdx(idx);
   307→        }
   308→      },
   309→    }),
   310→    [blocks]
   311→  );
   312→
   313→  // Jump to a heading by block index — used by the side drawer.
   314→  const jumpToHeading = useCallback((idx: number) => {
   315→    const el = blockRefs.current[idx];
   316→    if (el) {
   317→      el.scrollIntoView({ behavior: "smooth", block: "start" });
   318→      el.classList.add("block-flash");
   319→      setTimeout(() => el.classList.remove("block-flash"), 2200);
   320→      setActiveIdx(idx);
   321→      setActiveHeadingIdx(idx);
   322→    }
   323→  }, []);
   324→
   325→  // React to highlightToken (from outline click) — scroll & flash.
   326→  useEffect(() => {
   327→    if (!highlightToken || !blocks || blocks.length === 0) return;
   328→    const { quote, keywords } = highlightToken;
   329→    const idx = findBlockIndex(blocks, quote, keywords);
   330→    if (idx < 0) return;
   331→    const el = blockRefs.current[idx];
   332→    if (el) {
   333→      el.scrollIntoView({ behavior: "smooth", block: "center" });
   334→      el.classList.add("block-flash");
   335→      setTimeout(() => el.classList.remove("block-flash"), 2200);
   336→      setActiveIdx(idx);
   337→    }
   338→  }, [highlightToken, blocks]);
   339→
   340→  // Compute search matches whenever query or blocks change
   341→  useEffect(() => {
   342→    if (!searchQuery.trim() || !blocks) {
   343→      setSearchMatches([]);
   344→      setSearchCursor(0);
   345→      return;
   346→    }
   347→    const q = searchQuery.toLowerCase();
   348→    const idxs: number[] = [];
   349→    for (let i = 0; i < blocks.length; i++) {
   350→      const t = (
   351→        blocks[i].text ||
   352→        blocks[i].table_caption ||
   353→        (Array.isArray(blocks[i].chart_caption) ? blocks[i].chart_caption.join(" ") : "") ||
   354→        (Array.isArray(blocks[i].image_caption) ? blocks[i].image_caption.join(" ") : "") ||
   355→        ""
   356→      ).toLowerCase();
   357→      if (t.includes(q)) idxs.push(i);
   358→    }
   359→    setSearchMatches(idxs);
   360→    setSearchCursor(0);
   361→    if (idxs.length > 0) {
   362→      const el = blockRefs.current[idxs[0]];
   363→      if (el) {
   364→        el.scrollIntoView({ behavior: "smooth", block: "center" });
   365→        el.classList.add("block-flash");
   366→        setTimeout(() => el.classList.remove("block-flash"), 1500);
   367→        setActiveIdx(idxs[0]);
   368→      }
   369→    }
   370→  }, [searchQuery, blocks]);
   371→
   372→  const gotoMatch = (delta: number) => {
   373→    if (searchMatches.length === 0) return;
   374→    const next = (searchCursor + delta + searchMatches.length) % searchMatches.length;
   375→    setSearchCursor(next);
   376→    const idx = searchMatches[next];
   377→    const el = blockRefs.current[idx];
   378→    if (el) {
   379→      el.scrollIntoView({ behavior: "smooth", block: "center" });
   380→      el.classList.add("block-flash");
   381→      setTimeout(() => el.classList.remove("block-flash"), 1500);
   382→      setActiveIdx(idx);
   383→    }
   384→  };
   385→
   386→  // Keyboard shortcut: Ctrl+F / Cmd+F to open search, Esc to close
   387→  useEffect(() => {

   390→        e.preventDefault();

   392→      } else if (e.key === "Escape") {
   393→        setSearchOpen(false);
   394→        setSearchQuery("");
   395→      }
   396→    };
   397→    window.addEventListener("keydown", handler);
   398→    return () => window.removeEventListener("keydown", handler);
   399→  }, []);
   400→

   403→      {/* Status bar */}

   405→        <div className="px-3 py-2 bg-blue-50 border-b border-blue-200 text-xs text-blue-700 dark:bg-blue-950/30 dark:text-blue-300 flex items-center gap-2">
   406→          <Loader2 className="h-3.5 w-3.5 animate-spin" />

   408→        </div>
   409→      )}
   410→
   411→      {/* In-reader search bar */}
   412→      {searchOpen && (
   413→        <div className="px-3 py-2 bg-background border-b flex items-center gap-2">
   414→          <Search className="h-3.5 w-3.5 text-muted-foreground" />
   415→          <Input
   416→            value={searchQuery}
   417→            onChange={(e) => setSearchQuery(e.target.value)}
   418→            placeholder="查找段落…"
   419→            className="h-7 text-xs flex-1"
   420→            autoFocus
   421→          />
   422→          <span className="text-[11px] text-muted-foreground whitespace-nowrap">
   423→            {searchMatches.length > 0
   424→              ? `${searchCursor + 1}/${searchMatches.length}`
   425→              : searchQuery
   426→              ? "无匹配"
   427→              : "—"}
   428→          </span>
   429→          <Button
   430→            variant="ghost"
   431→            size="sm"
   432→            className="h-7 w-7 p-0"
   433→            onClick={() => gotoMatch(-1)}
   434→            disabled={searchMatches.length === 0}
   435→            title="上一个"
   436→          >
   437→            <ChevronUp className="h-3.5 w-3.5" />
   438→          </Button>
   439→          <Button
   440→            variant="ghost"
   441→            size="sm"
   442→            className="h-7 w-7 p-0"
   443→            onClick={() => gotoMatch(1)}
   444→            disabled={searchMatches.length === 0}
   445→            title="下一个"
   446→          >
   447→            <ChevronDown className="h-3.5 w-3.5" />
   448→          </Button>
   449→          <Button
   450→            variant="ghost"
   451→            size="sm"
   452→            className="h-7 w-7 p-0"
   453→            onClick={() => {
   454→              setSearchOpen(false);
   455→              setSearchQuery("");
   456→            }}
   457→            title="关闭 (Esc)"
   458→          >
   459→            <X className="h-3.5 w-3.5" />
   460→          </Button>
   461→        </div>
   462→      )}
   463→
   464→      {/* Toolbar with search toggle + headings drawer toggle */}
   465→      {!searchOpen && (
   466→        <div className="px-3 py-1 bg-background/60 border-b flex items-center justify-end gap-1">
   467→          <Button
   468→            variant="ghost"
   469→            size="sm"
   470→            className="h-7 gap-1 text-[11px] text-muted-foreground"
   471→            onClick={() => setHeadingsOpen(true)}
   472→            disabled={!hasBlocks || headings.length === 0}
   473→            title="段落导航"
   474→          >
   475→            <ListTree className="h-3 w-3" />
   476→            段落导航
   477→            {headings.length > 0 && (
   478→              <span className="ml-0.5 text-[9.5px] text-muted-foreground/70">
   479→                ({headings.length})
   480→              </span>
   481→            )}
   482→          </Button>
   483→          <Button
   484→            variant="ghost"
   485→            size="sm"
   486→            className="h-7 gap-1 text-[11px] text-muted-foreground"
   487→            onClick={() => setSearchOpen(true)}
   488→            title="查找 (Ctrl+F)"
   489→          >
   490→            <Search className="h-3 w-3" />
   491→            查找
   492→          </Button>
   493→        </div>
   494→      )}
   495→
   496→      {/* Heading navigator side drawer */}
   497→      <Sheet open={headingsOpen} onOpenChange={setHeadingsOpen}>
   498→        <SheetContent side="right" className="w-[360px] sm:w-[400px] p-0 flex flex-col">
   499→          <SheetHeader className="px-4 py-3 border-b">
   500→            <SheetTitle className="flex items-center gap-2 text-sm">
   501→              <ListTree className="h-4 w-4 text-sky-600" />
   502→              段落导航
   503→              <span className="ml-1 text-[11px] text-muted-foreground font-normal">
   504→                · {headings.length} 个标题
   505→              </span>
   506→            </SheetTitle>
   507→          </SheetHeader>
   508→          <div className="flex-1 overflow-y-auto scrollbar-thin px-2 py-2">
   509→            {headings.length === 0 ? (
   510→              <div className="text-center py-8 text-muted-foreground text-xs">
   511→                暂无标题结构
   512→                <div className="mt-1 text-[10px] text-muted-foreground/70">
   513→                  MinerU 解析完成后此处显示论文大小标题
   514→                </div>
   515→              </div>
   516→            ) : (

   929→// ---------- helpers ----------
   930→
   931→/**
   932→ * Find the index of the block whose text best matches the given quote or
   933→ * keywords. Uses a scoring approach:
   934→ *
   935→ *   - exact quote substring match  → score 1000
   936→ *   - quote-head (first 25 chars) substring match → score 800
   937→ *   - per-keyword substring match  → +100/keyword (longer keywords weighted more)
   938→ *   - prefer blocks where MORE keywords match
   939→ *
   940→ * Returns the highest-scoring block index, or -1 if no block scored > 0.
   941→ */
   942→function findBlockIndex(
   943→  blocks: MinerUBlock[],
   944→  quote: string,
   945→  keywords: string[]
   946→): number {
   947→  if (!blocks || blocks.length === 0) return -1;
   948→
   949→  // Normalize a string for matching: lowercase, collapse whitespace.
   950→  // Also strip <sup>/<sub>/<i> HTML tags so e.g. "SiglecF<sup>hi</sup>"
   951→  // matches a search for "siglecfhi".
   952→  const norm = (s: string) =>
   953→    (s || "")
   954→      .toLowerCase()
   955→      .replace(/<\/?(?:sup|sub|i|b|em|strong|span)[^>]*>/gi, "")
   956→      .replace(/\s+/g, " ")
   957→      .trim();
   958→
   959→  const q = norm(quote || "");
   960→  const qHead = q.slice(0, 25);
   961→  const kws = (keywords || [])
   962→    .filter((k) => k && k.length >= 2)
   963→    .map(norm)
   964→    .sort((a, b) => b.length - a.length); // longest first
   965→
   966→  // Extract any "Figure N" / "Fig N" / "Fig NA" references from the quote
   967→  // itself — if the LLM included the figure callout in its quote, we want
   968→  // blocks that mention the same figure to score higher.
   969→  const figRefsInQuote = new Set<string>();
   970→  const figRefRe = /\b(?:fig(?:ure|\.)?)\s*(\d+)\s*([a-z])?/gi;
   971→  let m: RegExpExecArray | null;
   972→  while ((m = figRefRe.exec(q)) !== null) {
   973→    const num = m[1];
   974→    const panel = m[2] || "";
   975→    figRefsInQuote.add(`figure ${num}`);
   976→    if (panel) {
   977→      figRefsInQuote.add(`figure ${num}${panel}`);
   978→      figRefsInQuote.add(`fig. ${num}`);
   979→      figRefsInQuote.add(`fig ${num}${panel}`);
   980→    }
   981→  }
   982→
   983→  let bestIdx = -1;
   984→  let bestScore = 0;
   985→
   986→  for (let i = 0; i < blocks.length; i++) {
   987→    const t = norm(blocks[i].text || "");
   988→    if (!t) continue;
   989→
   990→    let score = 0;
   991→
   992→    // 1) Exact quote match
   993→    if (q.length > 4 && t.includes(q)) {
   994→      score += 1000;
   995→    }
   996→    // 2) Quote head match (LLM often paraphrases the tail)
   997→    else if (qHead.length > 6 && t.includes(qHead)) {
   998→      score += 800;
   999→    }
  1000→
  1001→    // 3) Keyword matches — longer keywords score more
  1002→    for (const kw of kws) {
  1003→      if (t.includes(kw)) {
  1004→        // Weight by length: 4-char keyword = 100, 10-char = 250
  1005→        score += Math.min(300, 25 + kw.length * 25);
  1006→      }
  1007→    }
  1008→
  1009→    // 4) Token overlap bonus: split quote into words, count how many appear
  1010→    if (q.length > 4) {
  1011→      const tokens = q.split(/[,\s.;:()]+/).filter((w) => w.length > 3);
  1012→      let hits = 0;
  1013→      for (const tok of tokens) {
  1014→        if (t.includes(tok)) hits++;
  1015→      }
  1016→      if (tokens.length > 0) {
  1017→        score += Math.round((hits / tokens.length) * 200);
  1018→      }
  1019→    }
  1020→
  1021→    // 5) Figure-reference bonus — if the quote mentions "Figure 1A", boost
  1022→    // blocks that also mention "Figure 1" / "Fig. 1A". This is the panel
  1023→    // jump fallback when the LLM's quote is too short or paraphrased.
  1024→    if (figRefsInQuote.size > 0) {
  1025→      for (const ref of figRefsInQuote) {
  1026→        if (t.includes(ref)) {
  1027→          score += 150;
  1028→          break;
  1029→        }
  1030→      }
  1031→    }
  1032→
  1033→    if (score > bestScore) {
  1034→      bestScore = score;
  1035→      bestIdx = i;
  1036→    }
  1037→  }
  1038→
  1039→  // Require at least some signal — avoid jumping to block 0 on no match
  1040→  return bestScore > 0 ? bestIdx : -1;
  1041→}
  1042→
