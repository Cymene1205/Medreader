   167→  }, [paperId]);
   168→
   169→  const dismissLlmBanner = useCallback(() => {
   170→    setLlmBannerDismissed(true);

   172→      localStorage.setItem("medreader.llm.bannerDismissed", "1");

   174→      // ignore
   175→    }

   178→  const onFile = useCallback(
   179→    async (file: File) => {
   180→      if (!file) return;
   181→      if (!file.name.toLowerCase().endsWith(".pdf")) {
   182→        alert("目前仅支持 PDF 格式");
   183→        return;
   184→      }
   185→      const buf = await file.arrayBuffer();
   186→      const bufForText = buf.slice(0);
   187→      setFileData(buf);
   188→      setFileName(file.name);
   189→      setOutline(null);
   190→      setOutlineError(null);
   191→      setSelectedText("");
   192→      setSelectedBlockIdx(null);
   193→      setAttachedImage(null);
   194→      setActiveChildId(undefined);
   195→      setPaperMarkdown(null);
   196→      setPaperBlocks(null);
   197→      setPaperImagesDir(null);
   198→      // Reset Stage 2 state for the new upload
   199→      setFigures([]);
   200→      setCitations([]);
   201→      setFiguresStatus("idle");
   202→      setFiguresLoading(false);
   203→      // Left panel: 全文框架 starts expanded during loading (so the user
   204→      // sees the "analyzing…" / progress indicator), then auto-collapses
   205→      // once Stage 1 results arrive. Heading panel was removed from the
   206→      // left column — paragraph navigation is now a button inside the
   207→      // 智能解析 toolbar.
   208→      setOutlineCollapsed(false);
   209→      // Default the center view to PDF while the MinerU parse runs (it takes

   212→      // auto-switch to 智能解析 — unless the user has already picked a tab

   216→      setUploadStage("uploading");
   217→      setMineruStatus("上传中…");
   218→
   219→      // Upload to server — quota is checked server-side.
   220→      let serverPaperId: string | null = null;
   221→      let serverParsedText: string | null = null;
   222→      let serverMarkdown: string | null = null;
   223→      let serverBlocks: MinerUBlock[] | null = null;
   224→      let serverImagesDir: string | null = null;

   226→        const fd = new FormData();

   230→          const j = await upRes.json().catch(() => ({}));
   231→          throw new Error(j.error || "今日解析额度已用尽");
   232→        }
   233→        if (!upRes.ok) {
   234→          const j = await upRes.json().catch(() => ({}));
   235→          throw new Error(j.error || `Upload failed HTTP ${upRes.status}`);
   236→        }
   237→        const upData = await upRes.json();
   238→        serverPaperId = upData.paperId;
   239→        setPaperId(upData.paperId);
   240→        setUploadStage("parsing");
   241→        setMineruStatus("MinerU 解析中（30-90 秒）…");
   242→
   243→        // Poll for parse status
   244→        for (let i = 0; i < 120; i++) {
   245→          await new Promise((r) => setTimeout(r, 2000));
   246→          const sRes = await fetch(`/api/paper/${upData.paperId}`);
   247→          if (sRes.ok) {
   248→            const sData = await sRes.json();
   249→            if (sData.parseStatus === "done") {
   250→              serverParsedText = sData.parsedText;
   251→              serverMarkdown = sData.markdown;
   252→              serverBlocks = sData.blocks;
   253→              serverImagesDir = sData.imagesDir;
   254→              break;
   255→            }
   256→            if (sData.parseStatus === "error") {
   257→              throw new Error("MinerU 解析失败，已尝试 pdfjs 兜底");
   258→            }
   259→            if (i % 5 === 0) {
   260→              setMineruStatus(`MinerU 解析中…（已等 ${(i + 1) * 2}s）`);
   261→            }
   262→          }
   263→        }
   264→      } catch (e) {
   265→        setOutlineError(e instanceof Error ? e.message : String(e));
   266→        setUploadStage("idle");
   267→        setMineruStatus("");
   268→        return;
   269→      }
   270→
   271→      // Use server-parsed content if available; otherwise client extraction
   272→      let full = "";
   273→      if (serverParsedText) {
   274→        full = serverParsedText;
   275→      } else {
   276→        try {
   277→          const lib = await import("pdfjs-dist");
   278→          lib.GlobalWorkerOptions.workerSrc = `https://cdnjs.cloudflare.com/ajax/libs/pdf.js/${lib.version}/pdf.worker.min.mjs`;
   279→          const doc = await lib.getDocument({ data: new Uint8Array(bufForText) }).promise;
   280→          for (let i = 1; i <= doc.numPages; i++) {
   281→            const page = await doc.getPage(i);
   282→            const tc = await page.getTextContent();
   283→            const pageText = tc.items
   284→              .map((it: any) => (typeof it.str === "string" ? it.str : ""))
   285→              .join(" ");
   286→            full += `\n[Page ${i}]\n${pageText}\n`;
   287→            page.cleanup();
   288→          }
   289→        } catch (e) {
   290→          // ignore; we'll still try analyze with empty text
   291→        }
   292→      }
   293→
   294→      // Progressive: set plain text + markdown/blocks as soon as they're available.
   295→      setPaperText(full);
   296→      setPaperMarkdown(serverMarkdown);
   297→      setPaperBlocks(serverBlocks);
   298→      setPaperImagesDir(serverImagesDir);
   299→      setMineruStatus("");
   300→
   301→      // Auto-switch from the initial PDF view to 智能解析 now that the

   322→            text: full,
   323→            markdown: serverMarkdown || undefined,

   325→            paperId: serverPaperId,
   326→          }),
   327→        });

   331→        setUploadStage("done");

   336→        setOutlineCollapsed(true);
   337→      } catch (e) {
   338→        setOutlineError(e instanceof Error ? e.message : String(e));
   339→        setUploadStage("idle");
   340→      } finally {
   341→        setOutlineLoading(false);
   342→      }
   343→
   344→      // ── Stage 2: trigger figure analysis (Call A) in parallel with Stage 1.
   345→      // This runs AFTER MinerU's figure extraction (which happened during
   346→      // upload). Call A is one LLM call for all figures + writes back the
   347→      // argumentSpine. We poll for completion so the UI updates as figures
   348→      // become ready.
   349→      // ─────────────────────────────────────────────────────────────────
   350→      if (serverPaperId) {
   351→        setFiguresLoading(true);
   352→        setFiguresStatus("extracting");
   353→        // Fire-and-forget the figures POST (Call A). It writes to DB on
   354→        // completion; we poll the GET endpoint for results.
   355→        fetch("/api/figures", {
   356→          method: "POST",
   357→          headers: {
   358→            "Content-Type": "application/json",
   359→            ...refreshLLMHeaders(),
   360→          },
   361→          body: JSON.stringify({ paperId: serverPaperId }),
   362→        }).catch((e) => {
   363→          console.warn("[figures] Call A trigger failed:", e);
   364→          setFiguresStatus("error");
   365→        });
   366→
   367→        // Poll for figures + updated analysisJson + citations.
   368→        // We also drive the figuresStatus state machine here so the
   369→        // OutlinePanel 论证主线 progress indicator reflects the current
   370→        // sub-step (extracting → call-a → spine → done).
   371→        let pollCount = 0;
   372→        const poll = async () => {
   373→          pollCount++;
   374→          try {
   375→            const [figRes, analysisRes, paperRes] = await Promise.all([
   376→              fetch(`/api/figures?paperId=${serverPaperId}`).then((r) => r.json()),
   377→              fetch(`/api/analyze?paperId=${serverPaperId}`).then((r) => r.json()),
   378→              fetch(`/api/paper/${serverPaperId}`).then((r) => r.json()),
   379→            ]);
   380→            if (Array.isArray(figRes.figures)) {
   381→              setFigures(figRes.figures as Figure[]);
   382→            }
   383→            // Refresh outline if analysisJson has been updated (spine filled)
   384→            if (analysisRes.outline) {
   385→              setOutline(analysisRes.outline as Outline);
   386→            }
   387→            // Load citations (for panel chip click → quote jump)
   388→            if (Array.isArray(paperRes.citations)) {
   389→              setCitations(paperRes.citations as Citation[]);
   390→            }
   391→
   392→            // ── Drive the figuresStatus state machine ──
   393→            const figs: Figure[] = Array.isArray(figRes.figures) ? figRes.figures : [];
   394→            const allHaveQuestion =
   395→              figs.length > 0 && figs.every((f) => f.question);
   396→            const spineReady = !!analysisRes.outline?.argumentSpine;
   397→
   398→            if (spineReady && allHaveQuestion) {
   399→              setFiguresStatus("done");
   400→              setFiguresLoading(false);
   401→              return; // stop polling
   402→            }
   403→            if (allHaveQuestion && !spineReady) {
   404→              setFiguresStatus("spine");
   405→            } else if (figs.length > 0 && !allHaveQuestion) {
   406→              setFiguresStatus("call-a");
   407→            } else if (pollCount > 1) {
   408→              // After 2+ polls still no figures — likely Call A still
   409→              // processing or figure extraction slow. Stay in extracting.
   410→              setFiguresStatus("extracting");
   411→            }
   412→
   413→            // Safety: stop after ~6 minutes of polling (90 iterations × 4s)
   414→            if (pollCount > 90) {
   415→              console.warn("[poll] exhausted 90 iterations, giving up");
   416→              setFiguresStatus("error");
   417→              setFiguresLoading(false);
   418→              return;
   419→            }
   420→          } catch (e) {
   421→            console.warn("[poll] figures/analyze fetch failed:", e);
   422→          }
   423→          setTimeout(poll, 4000);
   424→        };
   425→        setTimeout(poll, 3000);
   426→      }
   427→    },
   428→    []
   429→  );

   485→  const stageLabel: Record<UploadStage, string> = {
   486→    idle: "",
   487→    uploading: "上传中…",
   488→    parsing: "MinerU 解析中…",
   489→    analyzing: "AI 分析中…",
   490→    done: "分析完成",
   491→  };
   492→
   493→  const isBusy = outlineLoading || uploadStage === "uploading" || uploadStage === "parsing";
   494→

   496→    <div className="h-screen w-screen flex flex-col bg-background overflow-hidden">
   497→      {/* Header */}
   498→      <header className="glass-header h-12 flex-shrink-0 flex items-center px-4 gap-3 border-b border-border/30">
   499→        <div className="flex items-center gap-2">
   500→          <div className="w-7 h-7 rounded-md bg-primary flex items-center justify-center shadow-sm">
   501→            <BookOpen className="h-4 w-4 text-primary-foreground" />
   502→          </div>
   503→          <span className="font-semibold text-sm text-background">MedReader Agent</span>
   504→          <span className="text-[10px] opacity-70 hidden sm:inline">
   505→            MinerU 驱动 · 智能解析
   506→          </span>
   507→        </div>
   508→
   509→        <div className="flex-1" />
   510→
   511→        {/* Upload stage indicator */}
   512→        {uploadStage !== "idle" && uploadStage !== "done" && (

   518→
   519→        {/* LLM settings button + warning if not configured */}
   520→        <Button

   522→          size="sm"
   523→          className={
